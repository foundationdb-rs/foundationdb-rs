57/// A key range reported by one of the conflict-range special keyspaces.
58///
59/// Returned by [`Transaction::conflicting_keys`] (the ranges that made a commit
60/// fail), [`Transaction::read_conflict_ranges`] and
61/// [`Transaction::write_conflict_ranges`] (the ranges the transaction has
62/// accumulated so far).
63///
64/// Those keyspaces all encode ranges using boundary markers:
65/// - Value `b"1"` marks the inclusive start of a range
66/// - Value `b"0"` marks the exclusive end of a range

71}

100/// Incremental parser of the `b"1"` / `b"0"` boundary encoding used by the
101/// `\xff\xff/transaction/*` conflict-range special keyspaces.
102///
103/// A range is only emitted once its end marker arrives, and the open begin
104/// marker survives across [`feed`](Self::feed) calls: a `b"1"` and its matching
105/// `b"0"` can land in two different `get_range` batches, so the parser is fed
106/// batch after batch and the caller paginates until the keyspace is exhausted.
107///
108/// A begin marker whose end marker never arrives is dropped: only complete
109/// ranges are returned, in every build profile.
110struct ConflictRangeParser {
111    prefix_len: usize,
112    ranges: Vec<ConflictRange>,
113    open_begin: Option<Vec<u8>>,
114}
115
116impl ConflictRangeParser {
117    /// `prefix` is the special keyspace prefix to strip from the keys fed in.
118    fn new(prefix: &[u8]) -> Self {
119        Self {
120            prefix_len: prefix.len(),
121            ranges: Vec::new(),
122            open_begin: None,
123        }
124    }
125
126    /// Feeds one batch of `(key, value)` pairs, in keyspace order.
127    fn feed<'a, I>(&mut self, batch: I)
128    where
129        I: IntoIterator<Item = (&'a [u8], &'a [u8])>,
130    {
131        for (raw_key, value) in batch {
132            let key = raw_key.get(self.prefix_len..).unwrap_or_default().to_vec();
133
134            match value {
135                b"1" => {
136                    #[cfg(feature = "trace")]
137                    if self.open_begin.is_some() {
138                        tracing::warn!(
139                            "'1' marker following an unpaired one in a conflict range keyspace, range dropped"
140                        );
141                    }
142
143                    self.open_begin = Some(key);
144                }
145                b"0" => {
146                    if let Some(begin) = self.open_begin.take() {
147                        self.ranges.push(ConflictRange { begin, end: key });
148                    }
149                }
150                _ => {}
151            }
152        }
153    }
154
155    /// Returns the complete ranges parsed so far, dropping a begin marker left
156    /// open by the end of the stream.
157    fn finish(self) -> Vec<ConflictRange> {
158        #[cfg(feature = "trace")]
159        if self.open_begin.is_some() {
160            tracing::warn!(
161                "unpaired '1' marker at the end of a conflict range keyspace, range dropped"
162            );
163        }
164
165        self.ranges
166    }
167}

173    tr: Transaction,

176impl TransactionCommitted {
177    /// Retrieves the database version number at which a given transaction was committed.
178    ///
179    /// Read-only transactions do not modify the database when committed and will have a committed
180    /// version of -1. Keep in mind that a transaction which reads keys and then sets them to their
181    /// current values may be optimized to a read-only transaction.
182    ///
183    /// Note that database versions are not necessarily unique to a given transaction and so cannot
184    /// be used to determine in what order two transactions completed. The only use for this
185    /// function is to manually enforce causal consistency when calling `set_read_version()` on
186    /// another subsequent transaction.
187    ///
188    /// Most applications will not call this function.
189    pub fn committed_version(&self) -> FdbResult<i64> {
190        let mut version: i64 = 0;
191        error::eval(unsafe {
192            fdb_sys::fdb_transaction_get_committed_version(self.tr.inner.as_ptr(), &mut version)
193        })?;
194        Ok(version)
195    }
196
197    /// Reset the transaction to its initial state.
198    ///
199    /// This will not affect previously committed data.
200    ///
201    /// This is similar to dropping the transaction and creating a new one.
202    /// User versions allocated before reset are invalid for the new attempt,
203    /// which starts allocating again at zero.
204    pub fn reset(mut self) -> Transaction {
205        self.tr.reset();
206        self.tr
207    }
208}
209impl From<TransactionCommitted> for Transaction {
210    fn from(tc: TransactionCommitted) -> Transaction {
211        tc.reset()
212    }

215/// A failed to commit transaction.
216pub struct TransactionCommitError {
217    tr: Transaction,
218    err: FdbError,
219}
220
221impl TransactionCommitError {
222    /// Implements the recommended retry and backoff behavior for a transaction. This function knows
223    /// which of the error codes generated by other `Transaction` functions represent temporary
224    /// error conditions and which represent application errors that should be handled by the
225    /// application. It also implements an exponential backoff strategy to avoid swamping the
226    /// database cluster with excessive retries when there is a high level of conflict between
227    /// transactions.
228    ///
229    /// You should not call this method most of the times and use `Database::transact` which
230    /// implements a retry loop strategy for you.
231    ///
232    /// On success the transaction enters a new attempt: its
233    /// [usage](Transaction::attempt_usage) restarts from zero, while its
234    /// [client budget](Transaction::set_client_budget) is kept. User versions
235    /// allocated by [`Transaction::allocate_user_version`] are invalid for the
236    /// new attempt, which starts allocating again at zero.
237    pub fn on_error(self) -> impl Future<Output = FdbResult<Transaction>> {
238        self.tr.mark_attempt_end();
239        let cause = self.err;
240
241        FdbFuture::<()>::new(unsafe {
242            fdb_sys::fdb_transaction_on_error(self.tr.inner.as_ptr(), self.err.code())
243        })
244        .map_ok(move |()| {
245            self.tr.end_attempt(AttemptOutcome::Retried { cause });
246            self.tr.begin_attempt_usage();
247            self.tr.reset_user_version_allocator();
248            self.tr
249        })
250    }
251
252    /// Reads the conflicting key ranges that caused this commit failure.
253    ///
254    /// Only returns meaningful results if
255    /// [`TransactionOption::ReportConflictingKeys`](crate::options::TransactionOption::ReportConflictingKeys)
256    /// was set on the transaction **and** the error is `not_committed` (code 1020).
257    ///
258    /// Must be called **before** [`on_error`](Self::on_error) which resets the transaction.
259    ///
260    /// # Errors
261    ///
262    /// Returns an `FdbError` if the special keyspace read fails.
263    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
264    pub async fn conflicting_keys(&self) -> FdbResult<Vec<ConflictingKeyRange>> {
265        self.tr.conflicting_keys().await
266    }
267
268    /// Reset the transaction to its initial state.
269    ///
270    /// This is similar to dropping the transaction and creating a new one.
271    /// User versions allocated before reset are invalid for the new attempt,
272    /// which starts allocating again at zero.
273    pub fn reset(mut self) -> Transaction {
274        self.tr.reset();
275        self.tr
276    }
277
278    /// Splits the error into the transaction it failed on and the error itself,
279    /// without resetting anything.
280    ///
281    /// Used by the retry runner to hand `on_error` an error chosen by a
282    /// [`RetryPolicy`](crate::runner::RetryPolicy) rather than the commit error
283    /// itself.
284    pub(crate) fn into_parts(self) -> (Transaction, FdbError) {
285        (self.tr, self.err)
286    }
287}

342/// In FoundationDB, a transaction is a mutable snapshot of a database.
343///
344/// All read and write operations on a transaction see and modify an otherwise-unchanging version of the database and only change the underlying database if and when the transaction is committed. Read operations do see the effects of previous write operations on the same transaction. Committing a transaction usually succeeds in the absence of conflicts.
345///
346/// Applications must provide error handling and an appropriate retry loop around the application code for a transaction. See the documentation for [fdb_transaction_on_error()](https://apple.github.io/foundationdb/api-c.html#transaction).
347///
348/// Transactions group operations into a unit with the properties of atomicity, isolation, and durability. Transactions also provide the ability to maintain an application’s invariants or integrity constraints, supporting the property of consistency. Together these properties are known as ACID.
349///
350/// Transactions are also causally consistent: once a transaction has been successfully committed, all subsequently created transactions will see the modifications made by it.
351#[derive(Debug)]
352pub struct Transaction {
353    // Order of fields should not be changed, because Rust drops field top-to-bottom, and
354    // transaction should be dropped before cluster.
355    inner: NonNull<fdb_sys::FDBTransaction>,
356    /// Metrics collector of the transaction, attached at creation or by a
357    /// runner hook, see [`Transaction::attach_metrics`]. Set at most once.
358    metrics: OnceLock<TransactionMetrics>,
359    /// Always-on accounting of the current attempt, see [`crate::budget`].
360    usage: UsageSlot,
361    /// Client-side limits applied to the current attempt. Unlike `usage`, they
362    /// are configuration: they survive `on_error` and `reset`.
363    budget: Mutex<ClientBudget>,
364    /// The next user version to issue for an incomplete versionstamp in the
365    /// current transaction attempt.
366    user_version: AtomicU32,
367}
368unsafe impl Send for Transaction {}
369unsafe impl Sync for Transaction {}

398/// `RangeOption` represents a query parameters for range scan query.
399///
400/// You can construct `RangeOption` easily:
401///
402/// ```
403/// use foundationdb::RangeOption;
404///
405/// let opt = RangeOption::from((b"begin".as_ref(), b"end".as_ref()));
406/// let opt: RangeOption = (b"begin".as_ref()..b"end".as_ref()).into();
407/// let opt = RangeOption {
408///     limit: Some(10),
409///     ..RangeOption::from((b"begin".as_ref(), b"end".as_ref()))
410/// };
411/// ```
412#[derive(Debug, Clone)]
413pub struct RangeOption<'a> {
414    /// The beginning of the range.
415    pub begin: KeySelector<'a>,
416    /// The end of the range.
417    pub end: KeySelector<'a>,
418    /// If non-zero, indicates the maximum number of key-value pairs to return.
419    /// `None` and `Some(0)` both mean unlimited.
420    pub limit: Option<usize>,
421    /// If non-zero, indicates a (soft) cap on the combined number of bytes of keys and values to
422    /// return for each item.
423    pub target_bytes: usize,
424    /// One of the options::StreamingMode values indicating how the caller would like the data in
425    /// the range returned.
426    pub mode: options::StreamingMode,
427    /// If true, key-value pairs will be returned in reverse lexicographical order beginning at
428    /// the end of the range.
429    pub reverse: bool,
430    #[doc(hidden)]
431    pub __non_exhaustive: std::marker::PhantomData<()>,
432}
433
434impl RangeOption<'_> {
435    /// Reverses the range direction.
436    pub fn rev(mut self) -> Self {
437        self.reverse = !self.reverse;
438        self
439    }
440
441    pub fn next_range(mut self, kvs: &FdbValues) -> Option<Self> {
442        if !kvs.more() {
443            return None;
444        }
445
446        let last = kvs.last()?;
447        let last_key = last.key();
448
449        if let Some(limit) = self.limit.as_mut().filter(|limit| **limit != 0) {
450            *limit = limit.saturating_sub(kvs.len());
451            if *limit == 0 {
452                return None;
453            }
454        }
455
456        if self.reverse {
457            self.end.make_first_greater_or_equal(last_key);
458        } else {
459            self.begin.make_first_greater_than(last_key);
460        }
461        Some(self)
462    }
463
464    #[cfg_api_versions(min = 710)]
465    pub(crate) fn next_mapped_range(mut self, kvs: &MappedKeyValues) -> Option<Self> {
466        if !kvs.more() {
467            return None;
468        }
469
470        let last = kvs.last()?;
471        let last_key = last.parent_key();
472
473        if let Some(limit) = self.limit.as_mut().filter(|limit| **limit != 0) {
474            *limit = limit.saturating_sub(kvs.len());
475            if *limit == 0 {
476                return None;
477            }
478        }
479
480        if self.reverse {
481            self.end.make_first_greater_or_equal(last_key);
482        } else {
483            self.begin.make_first_greater_than(last_key);
484        }
485        Some(self)
486    }
487}

562impl Transaction {
563    pub(crate) fn new(inner: NonNull<fdb_sys::FDBTransaction>) -> Self {
564        Self {
565            inner,
566            metrics: OnceLock::new(),
567            usage: UsageSlot::default(),
568            budget: Mutex::new(ClientBudget::default()),
569            user_version: AtomicU32::new(0),
570        }
571    }
572
573    /// Allocates a user version for an incomplete [`Versionstamp`](crate::tuple::Versionstamp).
574    ///
575    /// Values start at zero and increase through `65535`. Values issued by
576    /// this allocator are unique within the current transaction attempt, across
577    /// all handles that share that transaction. Explicit user versions supplied
578    /// to `Versionstamp::incomplete` remain supported, but can collide with
579    /// allocator-issued versions or with each other.
580    ///
581    /// The allocator performs no database operations and adds no conflict
582    /// ranges. It has no reservation mechanism: allocate a value when building
583    /// the versionstamped mutation that will use it.
584    ///
585    /// `reset` and a successful `on_error` start a new transaction attempt and
586    /// restart allocation at zero. A cancelled transaction retains its issued
587    /// values until it is reset. Versions from a previous attempt therefore
588    /// must not be reused in mutations for the new attempt.
589    ///
590    /// # Errors
591    ///
592    /// Returns [`FdbBindingError::UserVersionExhausted`] after all 65536 user
593    /// versions have been issued for this attempt.
594    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
595    pub fn allocate_user_version(&self) -> Result<u16, FdbBindingError> {
596        self.user_version
597            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |current| {
598                (current <= u16::MAX as u32).then_some(current + 1)
599            })
600            .map(|current| current as u16)
601            .map_err(|_| FdbBindingError::UserVersionExhausted)
602    }
603
604    /// Attaches a metrics collector to this transaction and opens its first
605    /// attempt on the accounting generation currently in use.
606    ///
607    /// A transaction collects metrics for at most one
608    /// [`TransactionMetrics`]: attaching a second one does nothing, the first
609    /// collector keeps receiving everything. This is what
610    /// [`MetricsHooks`](crate::runner::MetricsHooks) uses to wire itself onto a
611    /// transaction created by the runner.
612    pub(crate) fn attach_metrics(&self, metrics: &TransactionMetrics) {
613        if self.metrics.set(metrics.clone()).is_ok() {
614            self.open_metrics_attempt();
615        }
616    }
617
618    /// The metrics collector of the transaction, if one is attached.
619    fn metrics(&self) -> Option<&TransactionMetrics> {
620        self.metrics.get()
621    }
622
623    /// Points the metrics at the accounting generation of the current attempt,
624    /// so that whatever happens from now on is recorded there.
625    fn open_metrics_attempt(&self) {
626        if let Some(metrics) = self.metrics() {
627            metrics.begin_attempt(self.usage.current());
628        }
629    }
630
631    /// Freezes the duration of the current attempt: it is over, only the retry
632    /// machinery runs from here.
633    fn mark_attempt_end(&self) {
634        if let Some(metrics) = self.metrics() {
635            metrics.mark_attempt_end();
636        }
637    }
638
639    /// Pushes the current attempt to the metrics report.
640    fn end_attempt(&self, outcome: AttemptOutcome) {
641        if let Some(metrics) = self.metrics() {
642            metrics.finish_attempt(outcome);
643        }
644    }

646    /// The accounting generation an operation issued now must record into.
647    ///

694    /// Sets the client-side budget of this transaction.
695    ///
696    /// The limits are **not** enforced by FoundationDB and not enforced
697    /// automatically: they are checked when you call
698    /// [`check_client_budget`](Self::check_client_budget). See [`crate::budget`]
699    /// for what is counted and how precise it is.
700    ///
701    /// Setting a budget starts a fresh accounting generation, so the limits
702    /// apply to what happens from now on. They then survive `on_error` and
703    /// `reset`, and apply to each subsequent attempt with usage back at zero.
704    /// That generation, and every later one, is measured with the
705    /// [`clock`](ClientBudget::clock) of this budget.
706    ///
707    /// On an instrumented transaction, set the budget before doing anything
708    /// else: the new generation is also the one the metrics record into, so
709    /// operations performed before this call are not reported.
710    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
711    pub fn set_client_budget(&self, budget: ClientBudget) {
712        // One critical section: store the budget and take its clock out, so the
713        // generation below cannot be stamped with the clock of a budget that a
714        // concurrent call has already replaced.
715        let clock = {
716            let mut slot = self
717                .budget
718                .lock()
719                .unwrap_or_else(|poisoned| poisoned.into_inner());
720            *slot = budget;
721            slot.clock.clone()
722        };
723        self.begin_attempt_with_clock(clock);
724    }

738    ///
739    /// This is a synchronous, cheap check: a few atomic loads and one reading of
740    /// the [`clock`](ClientBudget::clock) of the budget, the wall clock by
741    /// default, no call to the database. Nothing calls it for you, so call it
742    /// between the operations of your transaction, typically inside a
743    /// [`Database::run`](crate::Database::run) closure where
744    /// [`FdbBindingError`](crate::FdbBindingError) makes `?` work:

746    /// ```

751    /// # let keys: Vec<Vec<u8>> = vec![];

756    ///             max_bytes_read: Some(1024 * 1024),
757    ///             ..ClientBudget::default()

777        let budget = self
778            .budget
779            .lock()
780            .unwrap_or_else(|poisoned| poisoned.into_inner())
781            .clone();

920    /// Modify the database snapshot represented by transaction to perform the operation indicated
921    /// by operationType with operand param to the value stored by the given key.
922    ///
923    /// An atomic operation is a single database command that carries out several logical steps:
924    /// reading the value of a key, performing a transformation on that value, and writing the
925    /// result. Different atomic operations perform different transformations. Like other database
926    /// operations, an atomic operation is used within a transaction; however, its use within a
927    /// transaction will not cause the transaction to conflict.
928    ///
929    /// Atomic operations do not expose the current value of the key to the client but simply send
930    /// the database the transformation to apply. In regard to conflict checking, an atomic
931    /// operation is equivalent to a write without a read. It can only cause other transactions
932    /// performing reads of the key to conflict.
933    ///
934    /// By combining these logical steps into a single, read-free operation, FoundationDB can
935    /// guarantee that the transaction will not conflict due to the operation. This makes atomic
936    /// operations ideal for operating on keys that are frequently modified. A common example is
937    /// the use of a key-value pair as a counter.
938    ///
939    /// # Warning
940    ///
941    /// If a transaction uses both an atomic operation and a strictly serializable read on the same
942    /// key, the benefits of using the atomic operation (for both conflict checking and performance)
943    /// are lost.
944    ///
945    /// Versionstamped operands are forwarded unchanged and must match the selected
946    /// runtime API, including when the client was initialized outside this crate.
947    /// Before API 520, `SetVersionstampedKey` consumes a two-byte offset suffix and
948    /// `SetVersionstampedValue` replaces the first ten bytes without an offset
949    /// suffix. API 520 and later use a four-byte offset suffix for both mutations.
950    /// Use [`crate::tuple::pack_with_versionstamp_for_key`] or
951    /// [`crate::tuple::Subspace::pack_with_versionstamp_for_key`] for tuple keys.
952    /// Tuple values containing an incomplete versionstamp require API 520 or later:
953    /// stamping at byte zero would overwrite the tuple's type code.
954    #[cfg_attr(
955        feature = "trace",
956        tracing::instrument(level = "debug", skip(self, key, param))
957    )]
958    pub fn atomic_op(&self, key: &[u8], param: &[u8], op_type: options::MutationType) {
959        unsafe {
960            fdb_sys::fdb_transaction_atomic_op(
961                self.inner.as_ptr(),
962                key.as_ptr(),
963                fdb_len(key.len(), "key"),
964                param.as_ptr(),
965                fdb_len(param.len(), "param"),
966                op_type.code(),
967            )
968        }
969
970        self.usage()
971            .record_atomic_op((key.len() + param.len()) as u64);
972    }

1135    /// `usage` is the accounting generation to record into, `None` to skip
1136    /// accounting entirely.
1137    fn get_range_impl(
1138        &self,
1139        opt: &RangeOption,
1140        iteration: usize,
1141        snapshot: bool,
1142        usage: Option<Arc<AttemptUsage>>,
1143    ) -> impl Future<Output = FdbResult<FdbValues>> + Send + Sync + Unpin + use<> {
1144        let begin = &opt.begin;
1145        let end = &opt.end;
1146        let key_begin = begin.key();
1147        let key_end = end.key();
1148
1149        FdbFuture::<FdbValues>::new(unsafe {
1150            fdb_sys::fdb_transaction_get_range(
1151                self.inner.as_ptr(),
1152                key_begin.as_ptr(),
1153                fdb_len(key_begin.len(), "key_begin"),
1154                fdb_bool(begin.or_equal()),
1155                begin.offset(),
1156                key_end.as_ptr(),
1157                fdb_len(key_end.len(), "key_end"),
1158                fdb_bool(end.or_equal()),
1159                end.offset(),
1160                fdb_limit(opt.limit.unwrap_or(0)),
1161                fdb_limit(opt.target_bytes),
1162                opt.mode.code(),
1163                fdb_iteration(iteration),
1164                fdb_bool(snapshot),
1165                fdb_bool(opt.reverse),
1166            )
1167        })
1168        .map(move |result| {
1169            if let Ok(values) = &result {
1170                let kv_fetched = values.len();
1171                let mut bytes_count = 0;
1172
1173                for key_value in values.as_ref() {
1174                    let key_len = key_value.key().len();
1175                    let value_len = key_value.value().len();
1176
1177                    bytes_count += (key_len + value_len) as u64
1178                }
1179
1180                if let Some(usage) = usage.as_ref() {
1181                    usage.record_get_range(bytes_count, kv_fetched as u64);
1182                }
1183            };
1184
1185            result
1186        })
1187    }

1189    /// Mapped Range is an experimental feature introduced in FDB 7.1.
1190    /// It is intended to improve the client throughput and reduce latency for querying data through a Subspace used as a "index".
1191    /// In such a case, querying records by scanning an index in relational databases can be
1192    /// translated to a GetRange request on the index entries followed up by multiple GetValue requests for the record entries in FDB.
1193    ///
1194    /// This method is allowing FoundationDB "follow up" a GetRange request with GetValue requests,
1195    /// this can happen in one request without additional back and forth. Considering the overhead
1196    /// of each request, this saves time and resources on serialization, deserialization, and network.
1197    ///
1198    /// A mapped request will:
1199    ///
1200    /// * Do a range query (same as a `Transaction.get_range` request) and get the result. We call it the primary query.
1201    /// * For each key-value pair in the primary query result, translate it to a `get_range` query and get the result. We call them secondary queries.
1202    /// * Put all results in a nested structure and return them.
1203    ///
1204    /// **WARNING** : This feature is considered experimental at this time. It is only allowed when
1205    /// using snapshot isolation AND disabling read-your-writes.
1206    ///
1207    /// More info can be found in the relevant [documentation](https://github.com/apple/foundationdb/wiki/Everything-about-GetMappedRange#input).
1208    ///
1209    /// This is the "raw" version, users are expected to use [Transaction::get_mapped_ranges]
1210    ///
1211    /// In the [attempt usage](Self::attempt_usage), a resolved batch counts as
1212    /// one `call_get_range` and as the bytes of the primary key-values plus the
1213    /// nested key-values returned by the secondary queries.
1214    #[cfg_api_versions(min = 710)]
1215    #[cfg_attr(
1216        feature = "trace",
1217        tracing::instrument(level = "debug", skip(self, opt, mapper, snapshot))
1218    )]
1219    pub fn get_mapped_range(
1220        &self,
1221        opt: &RangeOption,
1222        mapper: &[u8],
1223        iteration: usize,
1224        snapshot: bool,
1225    ) -> impl Future<Output = FdbResult<MappedKeyValues>> + Send + Sync + Unpin + use<> {
1226        let begin = &opt.begin;
1227        let end = &opt.end;
1228        let key_begin = begin.key();
1229        let key_end = end.key();
1230
1231        let usage = self.usage();
1232
1233        FdbFuture::<MappedKeyValues>::new(unsafe {

1278    /// Mapped Range is an experimental feature introduced in FDB 7.1.
1279    /// It is intended to improve the client throughput and reduce latency for querying data through a Subspace used as a "index".
1280    /// In such a case, querying records by scanning an index in relational databases can be
1281    /// translated to a GetRange request on the index entries followed up by multiple GetValue requests for the record entries in FDB.
1282    ///
1283    /// This method is allowing FoundationDB "follow up" a GetRange request with GetValue requests,
1284    /// this can happen in one request without additional back and forth. Considering the overhead
1285    /// of each request, this saves time and resources on serialization, deserialization, and network.
1286    ///
1287    /// A mapped request will:
1288    ///
1289    /// * Do a range query (same as a `Transaction.get_range` request) and get the result. We call it the primary query.
1290    /// * For each key-value pair in the primary query result, translate it to a `get_range` query and get the result. We call them secondary queries.
1291    /// * Put all results in a nested structure and return them.
1292    ///
1293    /// **WARNING** : This feature is considered experimental at this time. It is only allowed when
1294    /// using snapshot isolation AND disabling read-your-writes.
1295    ///
1296    /// More info can be found in the relevant [documentation](https://github.com/apple/foundationdb/wiki/Everything-about-GetMappedRange#input).
1297    #[cfg_api_versions(min = 710)]
1298    #[cfg_attr(
1299        feature = "trace",
1300        tracing::instrument(level = "debug", skip(self, opt, mapper, snapshot))
1301    )]
1302    pub fn get_mapped_ranges<'a>(
1303        &'a self,
1304        opt: RangeOption<'a>,
1305        mapper: &'a [u8],
1306        snapshot: bool,
1307    ) -> impl Stream<Item = FdbResult<MappedKeyValues>> + Send + Sync + Unpin + 'a {
1308        stream::unfold((1, Some(opt)), move |(iteration, maybe_opt)| {
1309            if let Some(opt) = maybe_opt {
1310                Either::Left(
1311                    self.get_mapped_range(&opt, mapper, iteration as usize, snapshot)
1312                        .map(move |maybe_values| {
1313                            let next_opt = match &maybe_values {
1314                                Ok(values) => opt.next_mapped_range(values),
1315                                Err(..) => None,
1316                            };
1317                            Some((maybe_values, (iteration + 1, next_opt)))
1318                        }),
1319                )
1320            } else {
1321                Either::Right(future::ready(None))
1322            }
1323        })
1324    }

1373    /// Attempts to commit the sets and clears previously applied to the database snapshot
1374    /// represented by transaction to the actual database.
1375    ///
1376    /// The commit may or may not succeed – in particular, if a conflicting transaction previously
1377    /// committed, then the commit must fail in order to preserve transactional isolation. If the
1378    /// commit does succeed, the transaction is durably committed to the database and all
1379    /// subsequently started transactions will observe its effects.
1380    ///
1381    /// It is not necessary to commit a read-only transaction – you can simply drop it.
1382    ///
1383    /// Callers will usually want to retry a transaction if the commit or a another method on the
1384    /// transaction returns a retryable error (see `on_error` and/or `Database::transact`).
1385    ///
1386    /// As with other client/server databases, in some failure scenarios a client may be unable to
1387    /// determine whether a transaction succeeded. In these cases, `Transaction::commit` will return
1388    /// an error and `is_maybe_committed()` will returns true on that error. The `on_error` function
1389    /// treats this error as retryable, so retry loops that don’t check for `is_maybe_committed()`
1390    /// could execute the transaction twice. In these cases, you must consider the idempotence of
1391    /// the transaction. For more information, see [Transactions with unknown results](https://apple.github.io/foundationdb/developer-guide.html#developer-guide-unknown-results).
1392    ///
1393    /// Normally, commit will wait for outstanding reads to return. However, if those reads were
1394    /// snapshot reads or the transaction option for disabling “read-your-writes” has been invoked,
1395    /// any outstanding reads will immediately return errors.
1396    ///
1397    /// On an instrumented transaction, the commit ends the current attempt: its
1398    /// duration is recorded whatever the result, and a successful commit pushes
1399    /// the attempt to the metrics report as
1400    /// [`AttemptOutcome::Committed`](crate::metrics::AttemptOutcome::Committed).
1401    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
1402    pub fn commit(self) -> impl Future<Output = TransactionResult> + Send + Sync + Unpin {
1403        let metrics = self.metrics().cloned();
1404        let started_at = Instant::now();
1405
1406        FdbFuture::<()>::new(unsafe { fdb_sys::fdb_transaction_commit(self.inner.as_ptr()) }).map(
1407            move |r| {
1408                if let Some(metrics) = &metrics {
1409                    metrics.record_commit(started_at.elapsed());
1410                }
1411                match r {
1412                    Ok(()) => {
1413                        self.end_attempt(AttemptOutcome::Committed);
1414                        Ok(TransactionCommitted { tr: self })
1415                    }
1416                    Err(err) => Err(TransactionCommitError { tr: self, err }),
1417                }
1418            },
1419        )
1420    }

1422    /// Implements the recommended retry and backoff behavior for a transaction. This function knows
1423    /// which of the error codes generated by other `Transaction` functions represent temporary
1424    /// error conditions and which represent application errors that should be handled by the
1425    /// application. It also implements an exponential backoff strategy to avoid swamping the
1426    /// database cluster with excessive retries when there is a high level of conflict between
1427    /// transactions.
1428    ///
1429    /// It is not necessary to call `reset()` when handling an error with `on_error()` since the
1430    /// transaction has already been reset.
1431    ///
1432    /// You should not call this method most of the times and use `Database::transact` which
1433    /// implements a retry loop strategy for you.
1434    ///
1435    /// On success the transaction enters a new attempt: its
1436    /// [usage](Self::attempt_usage) restarts from zero, while its
1437    /// [client budget](Self::set_client_budget) is kept. User versions
1438    /// allocated by [`Self::allocate_user_version`] are invalid for the new
1439    /// attempt, which starts allocating again at zero.
1440    ///
1441    /// On failure, the transaction is dropped. Use [`Self::on_error_with_transaction`]
1442    /// to retain ownership on both success and failure.
1443    pub fn on_error(
1444        self,
1445        err: FdbError,
1446    ) -> impl Future<Output = FdbResult<Transaction>> + Send + Sync + Unpin {
1447        self.on_error_with_transaction(err)
1448            .map(|(transaction, result)| result.map(|()| transaction))

1451    /// Applies the same retry and backoff behavior as [`Self::on_error`], returning
1452    /// ownership of the transaction even when the error cannot be retried.
1453    ///
1454    /// On success, the returned transaction has been reset for a new attempt.
1455    /// On failure, it retains the state left by the native error handler, which
1456    /// may be canceled. Call [`Self::reset`] before using it for another attempt.
1457    /// This method does not replace or explicitly reset the failed transaction.
1458    pub fn on_error_with_transaction(
1459        self,
1460        err: FdbError,
1461    ) -> impl Future<Output = (Transaction, FdbResult<()>)> + Send + Sync + Unpin {
1462        self.mark_attempt_end();

1467        .map(move |result| {
1468            if result.is_ok() {
1469                self.end_attempt(AttemptOutcome::Retried { cause: err });
1470                self.begin_attempt_usage();
1471                self.reset_user_version_allocator();
1472            }
1473            (self, result)
1474        })
1475    }

1709        match self.get(METADATA_VERSION_KEY, snapshot).await {
1710            Ok(Some(fdb_slice)) => {

1761    /// instrumented transaction, the attempt being recorded is abandoned rather
1762    /// than reported: it reached no conclusion. User versions allocated before
1763    /// reset are invalid for the new attempt, which starts allocating again at
1764    /// zero.
1765    pub fn reset(&mut self) {
1766        unsafe { fdb_sys::fdb_transaction_reset(self.inner.as_ptr()) }
1767        self.begin_attempt_usage();
1768        self.reset_user_version_allocator();
1769    }
1770
1771    fn reset_user_version_allocator(&self) {
1772        self.user_version.store(0, Ordering::Relaxed);
1773    }

1775    /// Reads the conflicting key ranges from the special keyspace after a commit conflict.
1776    ///
1777    /// This method reads from `\xff\xff/transaction/conflicting_keys/` and parses the
1778    /// boundary encoding where `b"1"` marks range starts and `b"0"` marks range ends.
1779    ///
1780    /// The special keyspace read is resolved client-side — no network round-trip to the
1781    /// cluster. The future still goes through the FDB network thread event loop, but the
1782    /// data comes from an in-memory map populated during the commit response. Returns
1783    /// an empty `Vec` if
1784    /// [`TransactionOption::ReportConflictingKeys`](crate::options::TransactionOption::ReportConflictingKeys)
1785    /// was not set.
1786    ///
1787    /// Requires API version 630 or later: the special keyspace does not exist on
1788    /// older versions, where this read fails.
1789    ///
1790    /// Only complete ranges are returned. A begin marker whose end marker never
1791    /// arrives is dropped, identically in debug and release builds.
1792    ///
1793    /// # Errors
1794    ///
1795    /// Returns an `FdbError` if the special keyspace read fails.
1796    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
1797    pub async fn conflicting_keys(&self) -> FdbResult<Vec<ConflictingKeyRange>> {
1798        self.read_conflict_range_keyspace(CONFLICTING_KEYS_PREFIX, CONFLICTING_KEYS_END)
1799            .await
1800    }

1802    /// Reads the read conflict ranges accumulated by this transaction so far.
1803    ///
1804    /// These are the ranges the resolver will check for conflicts at commit
1805    /// time: every key read by the transaction, plus the ranges added with
1806    /// [`add_conflict_range`](Self::add_conflict_range). A point read of `k`
1807    /// shows up as `k..k\x00`.
1808    ///
1809    /// <div class="warning">
1810    /// Reading this keyspace waits for every pending read of the transaction to
1811    /// settle, since a read that has not returned yet has not registered its
1812    /// conflict range. Called in the middle of a batch of concurrent reads, it
1813    /// is therefore as slow as the slowest of them.
1814    /// </div>
1815    ///
1816    /// The read itself is performed by the binding on your behalf: it is not
1817    /// accounted in the [attempt usage](Self::attempt_usage) and does not
1818    /// consume the [client budget](Self::set_client_budget).
1819    ///
1820    /// Only complete ranges are returned, see [`conflicting_keys`](Self::conflicting_keys).
1821    ///
1822    /// # Errors
1823    ///
1824    /// Returns an `FdbError` if the special keyspace read fails.
1825    #[cfg_api_versions(min = 630)]
1826    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
1827    pub async fn read_conflict_ranges(&self) -> FdbResult<Vec<ConflictRange>> {
1828        self.read_conflict_range_keyspace(READ_CONFLICT_RANGE_PREFIX, READ_CONFLICT_RANGE_END)
1829            .await
1830    }
1831
1832    /// Reads the write conflict ranges accumulated by this transaction so far.
1833    ///
1834    /// These are the ranges the transaction will conflict *other* transactions
1835    /// on: every key it wrote, plus the ranges added with
1836    /// [`add_conflict_range`](Self::add_conflict_range). A single `set` of `k`
1837    /// shows up as `k..k\x00`.
1838    ///
1839    /// <div class="warning">
1840    /// Before the commit, this is an approximate <em>superset</em> of the final
1841    /// write conflict ranges when the transaction uses versionstamped keys: the
1842    /// versionstamp is only known at commit time, so the incomplete key is
1843    /// reported with its placeholder bytes.
1844    /// </div>
1845    ///
1846    /// The read itself is performed by the binding on your behalf: it is not
1847    /// accounted in the [attempt usage](Self::attempt_usage) and does not
1848    /// consume the [client budget](Self::set_client_budget).
1849    ///
1850    /// Only complete ranges are returned, see [`conflicting_keys`](Self::conflicting_keys).
1851    ///
1852    /// # Errors
1853    ///
1854    /// Returns an `FdbError` if the special keyspace read fails.
1855    #[cfg_api_versions(min = 630)]
1856    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
1857    pub async fn write_conflict_ranges(&self) -> FdbResult<Vec<ConflictRange>> {
1858        self.read_conflict_range_keyspace(WRITE_CONFLICT_RANGE_PREFIX, WRITE_CONFLICT_RANGE_END)
1859            .await
1860    }

1862    /// Reads one conflict-range special keyspace to its end and parses it.
1863    ///
1864    /// The reads go through the unmetered path: they are performed by the
1865    /// binding itself and must not show up in the user's usage or budget. The
1866    /// keyspace is paginated because a marker pair can straddle two batches,
1867    /// which [`ConflictRangeParser`] handles by carrying the open begin marker
1868    /// from one batch to the next.
1869    async fn read_conflict_range_keyspace(
1870        &self,
1871        prefix: &[u8],
1872        end: &[u8],
1873    ) -> FdbResult<Vec<ConflictRange>> {
1874        let mut parser = ConflictRangeParser::new(prefix);
1875        let mut next = Some(RangeOption::from((prefix, end)));
1876        let mut iteration = 1;
1877
1878        while let Some(opt) = next {
1879            let values = self.get_range_unmetered(&opt, iteration, false).await?;
1880            parser.feed(values.iter().map(|kv| (kv.key(), kv.value())));
1881            next = opt.next_range(&values);
1882            iteration += 1;
1883        }
1884
1885        Ok(parser.finish())
1886    }

1888    /// Adds a conflict range to a transaction without performing the associated read or write.
1889    ///
1890    /// # Note
1891    ///
1892    /// Most applications will use the serializable isolation that transactions provide by default
1893    /// and will not need to manipulate conflict ranges.
1894    pub fn add_conflict_range(
1895        &self,
1896        begin: &[u8],
1897        end: &[u8],
1898        ty: options::ConflictRangeType,
1899    ) -> FdbResult<()> {
1900        error::eval(unsafe {
1901            fdb_sys::fdb_transaction_add_conflict_range(
1902                self.inner.as_ptr(),
1903                begin.as_ptr(),
1904                fdb_len(begin.len(), "begin"),
1905                end.as_ptr(),
1906                fdb_len(end.len(), "end"),
1907                ty.code(),
1908            )
1909        })
1910    }

1921/// A retryable transaction, generated by Database.run
1922#[derive(Clone)]
1923pub struct RetryableTransaction {
1924    inner: Arc<Transaction>,
1925}
1926
1927impl Deref for RetryableTransaction {
1928    type Target = Transaction;
1929    fn deref(&self) -> &Transaction {
1930        self.inner.deref()
1931    }
1932}
1933
1934impl RetryableTransaction {
1935    pub(crate) fn new(t: Transaction) -> RetryableTransaction {
1936        RetryableTransaction { inner: Arc::new(t) }
1937    }
1938
1939    pub(crate) fn take(self) -> Result<Transaction, FdbBindingError> {
1940        // checking weak references
1941        if Arc::weak_count(&self.inner) != 0 {
1942            return Err(FdbBindingError::ReferenceToTransactionKept);
1943        }
1944        Arc::try_unwrap(self.inner).map_err(|_| FdbBindingError::ReferenceToTransactionKept)
1945    }
1946
1947    pub(crate) async fn on_error(
1948        self,
1949        err: FdbError,
1950    ) -> Result<Result<RetryableTransaction, FdbError>, FdbBindingError> {
1951        Ok(self
1952            .take()?
1953            .on_error(err)
1954            .await
1955            .map(RetryableTransaction::new))
1956    }
1957
1958    #[cfg_attr(feature = "trace", tracing::instrument(level = "debug", skip(self)))]
1959    pub(crate) async fn commit(
1960        self,
1961    ) -> Result<Result<TransactionCommitted, TransactionCommitError>, FdbBindingError> {
1962        Ok(self.take()?.commit().await)
1963    }
