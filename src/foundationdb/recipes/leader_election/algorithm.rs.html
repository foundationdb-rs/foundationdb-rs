23const STATE_SCHEMA_VERSION: u64 = 1;

25#[derive(Debug, Clone, Default, PartialEq, Eq)]
26struct DurableState {
27    revision: u64,
28    owner: Option<ParticipantId>,
29    lease_duration: Option<Duration>,
30}

32async fn read_state<T>(txn: &T, key: &[u8]) -> Result<DurableState>
33where
34    T: Deref<Target = Transaction>,
35{
36    let Some(value) = txn.get(key, false).await? else {
37        return Ok(DurableState::default());
38    };
39
40    decode_state(&value)
41}
42
43fn decode_state(value: &[u8]) -> Result<DurableState> {
44    let (
45        schema_version,
46        revision,
47        has_owner,
48        owner,
49        has_lease_duration,
50        lease_duration_secs,
51        lease_duration_subsec_nanos,
52    ): (u64, u64, bool, String, bool, u64, u32) = unpack(value)?;
53    if schema_version != STATE_SCHEMA_VERSION {
54        return Err(LeaderElectionError::InvalidState(format!(
55            "unknown durable state schema version {schema_version}, expected {STATE_SCHEMA_VERSION}"
56        )));
57    }
58    let owner = if has_owner {
59        Some(ParticipantId::new(owner)?)
60    } else if owner.is_empty() {
61        None
62    } else {
63        return Err(LeaderElectionError::InvalidState(
64            "released state contains an owner ID".to_owned(),
65        ));
66    };
67    if !has_lease_duration && (lease_duration_secs != 0 || lease_duration_subsec_nanos != 0) {
68        return Err(LeaderElectionError::InvalidState(
69            "missing lease duration has non-zero fields".to_owned(),
70        ));
71    }
72    if has_lease_duration && lease_duration_subsec_nanos >= 1_000_000_000 {
73        return Err(LeaderElectionError::InvalidState(
74            "lease duration subsecond nanos is out of range".to_owned(),
75        ));
76    }
77    let lease_duration =
78        has_lease_duration.then(|| Duration::new(lease_duration_secs, lease_duration_subsec_nanos));
79    if revision == 0 && (owner.is_some() || lease_duration.is_some()) {
80        return Err(LeaderElectionError::InvalidState(
81            "zero revision state is not empty".to_owned(),
82        ));
83    }
84    if revision > 0 && lease_duration.is_none() {
85        return Err(LeaderElectionError::InvalidState(
86            "created state has no lease duration".to_owned(),
87        ));
88    }
89    if lease_duration == Some(Duration::ZERO) {
90        return Err(LeaderElectionError::InvalidState(
91            "persisted lease duration is zero".to_owned(),
92        ));
93    }
94    Ok(DurableState {
95        revision,
96        owner,
97        lease_duration,
98    })
99}

101fn write_state<T>(txn: &T, key: &[u8], state: &DurableState)
102where
103    T: Deref<Target = Transaction>,
104{
105    txn.set(key, &encode_state(state));
106}
107
108fn encode_state(state: &DurableState) -> Vec<u8> {
109    let owner = state.owner.as_ref().map_or("", ParticipantId::as_str);
110    let (lease_duration_secs, lease_duration_subsec_nanos) =
111        state.lease_duration.map_or((0, 0), |duration| {
112            (duration.as_secs(), duration.subsec_nanos())
113        });
114    pack(&(
115        STATE_SCHEMA_VERSION,
116        state.revision,
117        state.owner.is_some(),
118        owner,
119        state.lease_duration.is_some(),
120        lease_duration_secs,
121        lease_duration_subsec_nanos,
122    ))
123}
124
125fn next_revision(state: &DurableState) -> Result<u64> {
126    state
127        .revision
128        .checked_add(1)
129        .ok_or(LeaderElectionError::RevisionExhausted)
130}

132#[derive(Debug, Clone, Copy, PartialEq, Eq)]
133enum Decision {
134    Leader(PollTransition),
135    Follow {
136        timer: ObservationTimer,
137        reason: &'static str,
138    },
139}
140
141fn same_record(
142    state: &DurableState,
143    owner: &ParticipantId,
144    rank: Rank,
145    lease_duration: Duration,
146) -> bool {
147    state.owner.as_ref() == Some(owner)
148        && state.revision == rank.as_u64()
149        && state.lease_duration == Some(lease_duration)
150}
151
152fn decide(state: &DurableState, participant: &ParticipantId, input: &PollInput) -> Decision {
153    if state.owner.is_none() {
154        return Decision::Leader(PollTransition::Acquired);
155    }
156    match input {
157        PollInput::Leadership {
158            rank,
159            lease_duration,
160            elapsed,
161        } if same_record(state, participant, *rank, *lease_duration)
162            && elapsed < lease_duration =>
163        {
164            Decision::Leader(PollTransition::Renewed)
165        }
166        PollInput::Leadership { .. } => Decision::Follow {
167            timer: ObservationTimer::Reset,
168            reason: "leadership_not_renewable",
169        },
170        PollInput::Observation {
171            owner,
172            rank,
173            lease_duration,
174            elapsed,
175        } if same_record(state, owner, *rank, *lease_duration) => {
176            if elapsed >= lease_duration {
177                Decision::Leader(if owner == participant {
178                    PollTransition::Reacquired
179                } else {
180                    PollTransition::TookOver
181                })
182            } else {
183                Decision::Follow {
184                    timer: ObservationTimer::Preserve,
185                    reason: "observation",
186                }
187            }
188        }
189        PollInput::Observation { .. } => Decision::Follow {
190            timer: ObservationTimer::Reset,
191            reason: "observation",
192        },
193        PollInput::Unknown => Decision::Follow {
194            timer: ObservationTimer::Reset,
195            reason: "unknown",
196        },
197    }
198}

200fn leader_result(
201    state: &DurableState,
202    participant: &ParticipantId,
203    lease_duration: Duration,
204    transition: PollTransition,
205) -> Result<(PollResult, DurableState)> {
206    let revision = next_revision(state)?;
207    let next_state = DurableState {
208        revision,
209        owner: Some(participant.clone()),
210        lease_duration: Some(lease_duration),
211    };
212    let rank = Rank::from(revision);
213
214    #[cfg(feature = "trace")]
215    let action = match transition {
216        PollTransition::Acquired => "acquisition",
217        PollTransition::Renewed => "renewal",
218        PollTransition::TookOver => "takeover",
219        PollTransition::Reacquired => "reacquisition",
220        PollTransition::Followed => "follower",
221    };
222
223    #[cfg(feature = "trace")]
224    tracing::debug!(
225        poll_outcome = "leader",
226        poll_action = action,
227        participant = participant.as_str(),
228        revision,
229        takeover = transition == PollTransition::TookOver,
230        reacquisition = transition == PollTransition::Reacquired,
231        "leader-election poll staged in transaction"
232    );
233
234    Ok((
235        PollResult::new(
236            PollOutcome::Leader { rank, transition },
237            NextState::Leadership {
238                rank,
239                lease_duration,
240            },
241        ),
242        next_state,
243    ))
244}
245
246fn follower_result(
247    state: &DurableState,
248    timer: ObservationTimer,
249    _reason: &'static str,
250) -> Result<PollResult> {
251    let owner = state.owner.clone().ok_or_else(|| {
252        LeaderElectionError::InvalidState(
253            "follower result requested from released state".to_owned(),
254        )
255    })?;
256    let lease_duration = state.lease_duration.ok_or_else(|| {
257        LeaderElectionError::InvalidState(
258            "follower result requested without lease duration".to_owned(),
259        )
260    })?;
261    let rank = Rank::from(state.revision);
262
263    #[cfg(feature = "trace")]
264    tracing::debug!(
265        poll_outcome = "follower",
266        poll_reason = _reason,
267        owner = owner.as_str(),
268        revision = rank.as_u64(),
269        preserve_timer = timer == ObservationTimer::Preserve,
270        "leader-election poll observed owner"
271    );
272
273    let next = NextState::Observation {
274        owner: owner.clone(),
275        rank,
276        lease_duration,
277        timer,
278    };
279    let outcome = PollOutcome::Follower {
280        owner,
281        rank,
282        lease_duration,
283    };
284
285    Ok(PollResult::new(outcome, next))
286}

288/// Pure poll step: the result and, for a leader outcome, the state to write.
289fn step(
290    state: &DurableState,
291    participant: &ParticipantId,
292    lease_duration: Duration,
293    input: &PollInput,
294) -> Result<(PollResult, Option<DurableState>)> {
295    match decide(state, participant, input) {
296        Decision::Leader(transition) => {
297            let (result, next_state) =
298                leader_result(state, participant, lease_duration, transition)?;
299            Ok((result, Some(next_state)))
300        }
301        Decision::Follow { timer, reason } => Ok((follower_result(state, timer, reason)?, None)),
302    }
303}

305pub(crate) async fn poll<T>(
306    txn: &T,
307    subspace: &Subspace,
308    lease_duration: Duration,
309    participant: &ParticipantId,
310    input: &PollInput,
311) -> Result<PollResult>
312where
313    T: Deref<Target = Transaction>,
314{
315    let key = state_key(subspace);
316    let state = read_state(txn, &key).await?;
317    let (result, next_state) = step(&state, participant, lease_duration, input)?;
318    if let Some(next_state) = next_state {
319        write_state(txn, &key, &next_state);
320    }
321    Ok(result)
322}

324pub(crate) async fn state<T>(txn: &T, subspace: &Subspace) -> Result<ElectionState>
325where
326    T: Deref<Target = Transaction>,
327{
328    let state = read_state(txn, &state_key(subspace)).await?;
329    Ok(ElectionState::new(
330        state.owner,
331        Rank::from(state.revision),
332        state.lease_duration,
333    ))
334}

336pub(crate) async fn resign<T>(
337    txn: &T,
338    subspace: &Subspace,
339    participant: &ParticipantId,
340    rank: Rank,
341    lease_duration: Duration,
342) -> Result<ResignOutcome>
343where
344    T: Deref<Target = Transaction>,
345{
346    let key = state_key(subspace);
347    let state = read_state(txn, &key).await?;
348    if !same_record(&state, participant, rank, lease_duration) {
349        #[cfg(feature = "trace")]
350        let rejection_reason = if state.owner.as_ref() != Some(participant) {
351            "owner_changed"
352        } else if state.revision != rank.as_u64() {
353            "revision_changed"
354        } else {
355            "lease_duration_changed"
356        };
357        #[cfg(feature = "trace")]
358        tracing::debug!(
359            resign_outcome = "rejected",
360            resign_reason = rejection_reason,
361            participant = participant.as_str(),
362            leadership_revision = rank.as_u64(),
363            revision = state.revision,
364            "leader-election stale resignation rejected"
365        );
366        return Ok(ResignOutcome::Rejected);
367    }
368
369    let released = DurableState {
370        revision: state.revision,
371        owner: None,
372        lease_duration: state.lease_duration,
373    };
374    write_state(txn, &key, &released);
375    #[cfg(feature = "trace")]
376    tracing::debug!(
377        resign_outcome = "resigned",
378        revision = released.revision,
379        "leader-election resignation staged in transaction"
380    );
381    Ok(ResignOutcome::Resigned)
382}

631    /// The pre-clock-free poll logic, copied verbatim from the parent of the
632    /// `decide` extraction and made pure: it computes the would-be written
633    /// state instead of staging it, and adopts the pending state directly.
634    mod legacy {
635        use super::*;
636
637        #[derive(Debug, Clone, PartialEq, Eq)]
638        pub(super) struct Observation {
639            pub(super) owner: ParticipantId,
640            pub(super) rank: Rank,
641            pub(super) lease_duration: Duration,
642            pub(super) first_observed_at: Duration,
643        }
644
645        #[derive(Debug, Clone, PartialEq, Eq)]
646        pub(super) struct Leadership {
647            pub(super) participant: ParticipantId,
648            pub(super) rank: Rank,
649            pub(super) lease_duration: Duration,
650            pub(super) last_renewed_at: Duration,
651        }
652
653        #[derive(Debug, Clone, PartialEq, Eq)]
654        pub(super) enum LocalState {
655            Unknown,
656            Observation(Observation),
657            Leadership(Leadership),
658        }
659
660        enum PendingNextState {
661            PreservedObservation(Observation),
662            NewObservation {
663                owner: ParticipantId,
664                rank: Rank,
665                lease_duration: Duration,
666            },
667            Leadership(Leadership),
668        }
669
670        impl PendingNextState {
671            fn into_local_state(self, adopted_at: Duration) -> LocalState {
672                match self {
673                    Self::PreservedObservation(observation) => LocalState::Observation(observation),
674                    Self::NewObservation {
675                        owner,
676                        rank,
677                        lease_duration,
678                    } => LocalState::Observation(Observation {
679                        owner,
680                        rank,
681                        lease_duration,
682                        first_observed_at: adopted_at,
683                    }),
684                    Self::Leadership(leadership) => LocalState::Leadership(leadership),
685                }
686            }
687        }
688
689        fn same_observation(state: &DurableState, observation: &Observation) -> bool {
690            state.owner.as_ref() == Some(&observation.owner)
691                && state.revision == observation.rank.as_u64()
692                && state.lease_duration == Some(observation.lease_duration)
693        }
694
695        fn valid_leadership(
696            state: &DurableState,
697            participant: &ParticipantId,
698            leadership: &Leadership,
699            now: Duration,
700        ) -> bool {
701            &leadership.participant == participant
702                && state.owner.as_ref() == Some(participant)
703                && state.revision == leadership.rank.as_u64()
704                && state.lease_duration == Some(leadership.lease_duration)
705                && now.saturating_sub(leadership.last_renewed_at) < leadership.lease_duration
706        }
707
708        fn leader_result(
709            state: &DurableState,
710            participant: &ParticipantId,
711            lease_duration: Duration,
712            attempt_started_at: Duration,
713            transition: PollTransition,
714        ) -> Result<(PollOutcome, PendingNextState)> {
715            let rank = Rank::from(next_revision(state)?);
716            Ok((
717                PollOutcome::Leader { rank, transition },
718                PendingNextState::Leadership(Leadership {
719                    participant: participant.clone(),
720                    rank,
721                    lease_duration,
722                    last_renewed_at: attempt_started_at,
723                }),
724            ))
725        }
726
727        fn follower_result(
728            state: &DurableState,
729            previous: Option<&Observation>,
730        ) -> Result<(PollOutcome, PendingNextState)> {
731            let owner = state.owner.clone().unwrap();
732            let lease_duration = state.lease_duration.unwrap();
733            let rank = Rank::from(state.revision);
734            let next_observation = match previous {
735                Some(previous) if same_observation(state, previous) => {
736                    PendingNextState::PreservedObservation(previous.clone())
737                }
738                Some(_) | None => PendingNextState::NewObservation {
739                    owner: owner.clone(),
740                    rank,
741                    lease_duration,
742                },
743            };
744            Ok((
745                PollOutcome::Follower {
746                    owner,
747                    rank,
748                    lease_duration,
749                },
750                next_observation,
751            ))
752        }

754        pub(super) fn step(
755            state: &DurableState,
756            lease_duration: Duration,
757            participant: &ParticipantId,
758            local_state: &LocalState,
759            attempt_started_at: Duration,
760            adopted_at: Duration,
761        ) -> Result<(PollOutcome, LocalState)> {
762            let (outcome, pending) = match local_state {
763                LocalState::Leadership(leadership)
764                    if valid_leadership(state, participant, leadership, attempt_started_at) =>
765                {
766                    leader_result(
767                        state,
768                        participant,
769                        lease_duration,
770                        attempt_started_at,
771                        PollTransition::Renewed,
772                    )
773                }
774                LocalState::Observation(observation) if state.owner.is_some() => {
775                    if same_observation(state, observation)
776                        && attempt_started_at.saturating_sub(observation.first_observed_at)
777                            >= observation.lease_duration
778                    {
779                        let transition = if &observation.owner == participant {
780                            PollTransition::Reacquired
781                        } else {
782                            PollTransition::TookOver
783                        };
784                        leader_result(
785                            state,
786                            participant,
787                            lease_duration,
788                            attempt_started_at,
789                            transition,
790                        )
791                    } else {
792                        follower_result(state, Some(observation))
793                    }
794                }
795                LocalState::Unknown if state.owner.is_some() => follower_result(state, None),
796                LocalState::Leadership(_) if state.owner.is_some() => follower_result(state, None),
797                LocalState::Observation(_) | LocalState::Unknown | LocalState::Leadership(_) => {
798                    leader_result(
799                        state,
800                        participant,
801                        lease_duration,
802                        attempt_started_at,
803                        PollTransition::Acquired,
804                    )
805                }
806            }?;
807            Ok((outcome, pending.into_local_state(adopted_at)))
808        }
809    }

811    /// Maps a legacy local state to the new helper. A leadership owned by
812    /// another participant is not expressible and claimed nothing: `Unknown`.
813    fn to_new(local: &legacy::LocalState, participant: &ParticipantId) -> LocalState {
814        match local {
815            legacy::LocalState::Unknown => LocalState::Unknown,
816            legacy::LocalState::Observation(observation) => {
817                LocalState::Observation(Observation::new(
818                    observation.owner.clone(),
819                    observation.rank,
820                    observation.lease_duration,
821                    observation.first_observed_at,
822                ))
823            }
824            legacy::LocalState::Leadership(leadership)
825                if &leadership.participant != participant =>
826            {
827                LocalState::Unknown
828            }
829            legacy::LocalState::Leadership(leadership) => LocalState::Leadership(Leadership::new(
830                leadership.rank,
831                leadership.lease_duration,
832                leadership.last_renewed_at,
833            )),
834        }
835    }

837    /// Compares a resulting local state; a resulting legacy leadership is
838    /// always owned by the poller, so only its rank, duration, and anchor count.
839    fn same_local(new: &LocalState, legacy: &legacy::LocalState) -> bool {
840        match (new, legacy) {
841            (LocalState::Unknown, legacy::LocalState::Unknown) => true,
842            (LocalState::Observation(new), legacy::LocalState::Observation(legacy)) => {
843                new.owner() == &legacy.owner
844                    && new.rank() == legacy.rank
845                    && new.lease_duration() == legacy.lease_duration
846                    && new.first_observed_at() == legacy.first_observed_at
847            }
848            (LocalState::Leadership(new), legacy::LocalState::Leadership(legacy)) => {
849                new.rank() == legacy.rank
850                    && new.lease_duration() == legacy.lease_duration
851                    && new.last_renewed_at() == legacy.last_renewed_at
852            }
853            _ => false,
854        }
855    }

857    #[test]
858    fn clock_free_poll_matches_legacy_poll_exhaustively() {
859        let secs = Duration::from_secs;
860        let participants = [id("a"), id("b")];
861        let durations = [secs(1), secs(2)];
862        let anchors: Vec<Duration> = (0..=5).map(secs).collect();
863
864        let mut states = Vec::new();
865        for revision in (0..=3).chain([u64::MAX - 1, u64::MAX]) {
866            for owner in [None, Some(id("a")), Some(id("b"))] {
867                for lease_duration in durations {
868                    let state = match (revision, &owner) {
869                        (0, None) => DurableState::default(),
870                        (0, Some(_)) => continue,
871                        _ => DurableState {
872                            revision,
873                            owner: owner.clone(),
874                            lease_duration: Some(lease_duration),
875                        },
876                    };
877                    if !states.contains(&state) {
878                        states.push(state);
879                    }
880                }
881            }
882        }
883
884        let mut cases = 0_usize;
885        let mut exhausted = 0_usize;
886        for participant in &participants {
887            let mut locals = vec![legacy::LocalState::Unknown];
888            for rank in (0..=3).chain([u64::MAX - 1, u64::MAX]).map(Rank::from) {
889                for lease_duration in durations {
890                    for &anchor in &anchors {
891                        for owner in &participants {
892                            locals.push(legacy::LocalState::Observation(legacy::Observation {
893                                owner: owner.clone(),
894                                rank,
895                                lease_duration,
896                                first_observed_at: anchor,
897                            }));
898                            // Legacy leadership tokens: the poller's own and a foreign one.
899                            locals.push(legacy::LocalState::Leadership(legacy::Leadership {
900                                participant: owner.clone(),
901                                rank,
902                                lease_duration,
903                                last_renewed_at: anchor,
904                            }));
905                        }
906                    }
907                }
908            }
909
910            for state in &states {
911                for local in &locals {
912                    let new_local = to_new(local, participant);
913                    for handle_lease in durations {
914                        for now in (0..=6).map(secs) {
915                            let adopted_at = now + secs(1);
916                            let legacy = legacy::step(
917                                state,
918                                handle_lease,
919                                participant,
920                                local,
921                                now,
922                                adopted_at,
923                            );
924                            let new = step(state, participant, handle_lease, &new_local.input(now));
925                            let context = format!(
926                                "state {state:?}, participant {participant:?}, local {local:?}, \
927                                 handle lease {handle_lease:?}, now {now:?}"
928                            );
929                            let ((legacy_outcome, legacy_next), (result, written)) =
930                                match (legacy, new) {
931                                    (Ok(legacy), Ok(new)) => (legacy, new),
932                                    (
933                                        Err(LeaderElectionError::RevisionExhausted),
934                                        Err(LeaderElectionError::RevisionExhausted),
935                                    ) => {
936                                        exhausted += 1;
937                                        cases += 1;
938                                        continue;
939                                    }
940                                    (legacy, new) => {
941                                        panic!("{context}: legacy {legacy:?}, new {new:?}")
942                                    }
943                                };
944                            let new_next = new_local.clone().adopt(now, adopted_at, result.next());
945
946                            assert_eq!(result.outcome(), &legacy_outcome, "{context}");
947                            assert!(same_local(&new_next, &legacy_next), "{context}");
948                            let expected_written =
949                                legacy_outcome.is_leader().then(|| DurableState {
950                                    revision: legacy_outcome.rank().as_u64(),
951                                    owner: Some(participant.clone()),
952                                    lease_duration: Some(handle_lease),
953                                });
954                            assert_eq!(written, expected_written, "{context}");
955                            cases += 1;
956                        }
957                    }
958                }
959            }
960        }
961        assert!(cases > 100_000, "only {cases} cases");
962        assert!(exhausted > 0, "revision exhaustion was not exercised");
963    }
