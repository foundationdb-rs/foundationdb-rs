1//! Decoding of reassembled profiling blobs into typed [`Event`]s.
2//!
3//! A blob is what the C++ client writes with `BinaryWriter(IncludeVersion())` for one sampled
4//! transaction: a little-endian `u64` protocol version followed by events serialized back to
5//! back. Each event starts with its `EventType` as a little-endian `i32`, then the shared
6//! `Event` fields and the event-specific ones. Byte strings are a little-endian `u32` length
7//! followed by the bytes, `Optional<T>` is a presence byte followed by `T`, vectors are a
8//! little-endian `u32` count followed by the elements.
9//!
10//! Reference: `fdbclient/include/fdbclient/ClientLogEvents.h` and `CommitTransactionRef` /
11//! `MutationRef::serialize` in `fdbclient/include/fdbclient/CommitTransaction.h`.
12
13use crate::event::{
14    Commit, CommitError, CommitRequest, Event, EventHeader, Get, GetError, GetRange, GetRangeError,
15    GetVersion, KeyRange, Mutation, ProtocolVersion, SpanContext,
16};
17use tracing::instrument;

19/// Errors returned by [`decode_events`].
20#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
21#[non_exhaustive]
22pub enum DecodeError {
23    /// The blob ended before a field could be read.
24    #[error("unexpected end of blob at offset {offset}: needed {needed} bytes, {remaining} left")]
25    UnexpectedEof {
26        /// Offset in the blob where the read started.
27        offset: usize,
28        /// Number of bytes the read needed.
29        needed: usize,
30        /// Number of bytes left in the blob.
31        remaining: usize,
32    },
33    /// The blob was written with a protocol version this crate does not decode.
34    #[error("unsupported protocol version {0:#018x}")]
35    UnsupportedProtocolVersion(u64),
36    /// An event carries a type code unknown to this crate.
37    #[error("unknown event type {0}")]
38    UnknownEventType(i32),
39    /// A conflict range uses the single-key encoding with a begin key not ending in `\x00`.
40    #[error("malformed key range at offset {offset}")]
41    MalformedKeyRange {
42        /// Offset in the blob where the key range starts.
43        offset: usize,
44    },
45    /// A mutation's checksum suffix could not be removed (`param2` shorter than the
46    /// checksum, and accumulative checksum index when present).
47    #[error("malformed mutation at offset {offset}")]
48    MalformedMutation {
49        /// Offset in the blob where the mutation starts.
50        offset: usize,
51    },
52}
53
54const EVENT_GET_VERSION: i32 = 0;
55const EVENT_GET: i32 = 1;
56const EVENT_GET_RANGE: i32 = 2;
57const EVENT_COMMIT: i32 = 3;
58const EVENT_ERROR_GET: i32 = 4;
59const EVENT_ERROR_GET_RANGE: i32 = 5;
60const EVENT_ERROR_COMMIT: i32 = 6;
61
62const CHECKSUM_FLAG_MASK: u8 = 0x80;
63const ACCUMULATIVE_CHECKSUM_INDEX_FLAG_MASK: u8 = 0x40;
64const CHECKSUM_LEN: usize = 4;
65const ACCUMULATIVE_CHECKSUM_INDEX_LEN: usize = 2;
66
67/// Minimum encoded size of a key range (two empty byte strings).
68const MIN_KEY_RANGE_LEN: usize = 8;
69/// Minimum encoded size of a mutation (type byte plus two empty byte strings).
70const MIN_MUTATION_LEN: usize = 9;
71
72/// Decodes one reassembled profiling blob (all chunks of one sampled transaction,
73/// concatenated) into its protocol version and events, in write order.
74///
75/// Only blobs written by 7.1 to 7.4 clients are supported. Never panics on malformed input,
76/// and allocations are bounded by the blob length.
77#[instrument(level = "trace", skip_all, fields(blob_len = blob.len()))]
78pub fn decode_events(blob: &[u8]) -> Result<(ProtocolVersion, Vec<Event>), DecodeError> {
79    let mut reader = Reader::new(blob);
80    let version = ProtocolVersion(reader.u64()?);
81    if !is_supported(version) {
82        return Err(DecodeError::UnsupportedProtocolVersion(version.0));
83    }
84    let mut events = Vec::new();
85    while !reader.is_empty() {
86        events.push(reader.event(version)?);
87    }
88    Ok((version, events))
89}
90
91fn is_supported(version: ProtocolVersion) -> bool {
92    matches!(
93        version,
94        ProtocolVersion::V7_1
95            | ProtocolVersion::V7_2
96            | ProtocolVersion::V7_3
97            | ProtocolVersion::V7_4
98    )
99}

101/// 7.2 switched the commit span context from `Optional<UID>` to `Optional<SpanContext>`
102/// (`hasOTELSpanContext`).
103fn has_otel_span_context(version: ProtocolVersion) -> bool {
104    version.0 >= ProtocolVersion::V7_2.0
105}

107struct Reader<'a> {
108    buf: &'a [u8],
109    pos: usize,
110}
111
112impl<'a> Reader<'a> {
113    fn new(buf: &'a [u8]) -> Self {
114        Reader { buf, pos: 0 }
115    }
116
117    fn remaining(&self) -> usize {
118        self.buf.len() - self.pos
119    }
120
121    fn is_empty(&self) -> bool {
122        self.remaining() == 0
123    }
124
125    fn take(&mut self, n: usize) -> Result<&'a [u8], DecodeError> {
126        match self.buf.get(self.pos..).and_then(|rest| rest.get(..n)) {
127            Some(bytes) => {
128                self.pos += n;
129                Ok(bytes)
130            }
131            None => Err(DecodeError::UnexpectedEof {
132                offset: self.pos,
133                needed: n,
134                remaining: self.remaining(),
135            }),
136        }
137    }
138
139    fn array<const N: usize>(&mut self) -> Result<[u8; N], DecodeError> {
140        let mut out = [0u8; N];
141        out.copy_from_slice(self.take(N)?);
142        Ok(out)
143    }
144
145    fn u8(&mut self) -> Result<u8, DecodeError> {
146        Ok(self.array::<1>()?[0])
147    }
148
149    fn bool(&mut self) -> Result<bool, DecodeError> {
150        Ok(self.u8()? != 0)
151    }
152
153    fn u32(&mut self) -> Result<u32, DecodeError> {
154        Ok(u32::from_le_bytes(self.array()?))
155    }
156
157    fn i32(&mut self) -> Result<i32, DecodeError> {
158        Ok(i32::from_le_bytes(self.array()?))
159    }
160
161    fn u64(&mut self) -> Result<u64, DecodeError> {
162        Ok(u64::from_le_bytes(self.array()?))
163    }
164
165    fn i64(&mut self) -> Result<i64, DecodeError> {
166        Ok(i64::from_le_bytes(self.array()?))
167    }
168
169    fn f64(&mut self) -> Result<f64, DecodeError> {
170        Ok(f64::from_le_bytes(self.array()?))
171    }
172
173    /// Length-prefixed byte string. The length is checked against the remaining bytes
174    /// before anything is allocated.
175    fn bytes(&mut self) -> Result<Vec<u8>, DecodeError> {
176        let len = self.u32()? as usize;
177        Ok(self.take(len)?.to_vec())
178    }
179
180    fn key_range(&mut self) -> Result<KeyRange, DecodeError> {
181        Ok(KeyRange {
182            begin: self.bytes()?,
183            end: self.bytes()?,
184        })
185    }
186
187    /// Count-prefixed vector. The initial capacity is capped by what the remaining bytes can
188    /// hold, so a garbage count cannot trigger a huge allocation.
189    fn vec<T>(
190        &mut self,
191        min_elem_len: usize,
192        mut elem: impl FnMut(&mut Self) -> Result<T, DecodeError>,
193    ) -> Result<Vec<T>, DecodeError> {
194        let count = self.u32()? as usize;
195        let mut out = Vec::with_capacity(count.min(self.remaining() / min_elem_len));
196        for _ in 0..count {
197            out.push(elem(self)?);
198        }
199        Ok(out)
200    }
201
202    fn header(&mut self) -> Result<EventHeader, DecodeError> {
203        let start_timestamp = self.f64()?;
204        let dc_id = self.bytes()?;
205        let tenant = if self.bool()? {
206            Some(self.bytes()?)
207        } else {
208            None
209        };
210        Ok(EventHeader {
211            start_timestamp,
212            dc_id,
213            tenant,
214        })
215    }
216
217    fn event(&mut self, version: ProtocolVersion) -> Result<Event, DecodeError> {
218        let event_type = self.i32()?;
219        if !(EVENT_GET_VERSION..=EVENT_ERROR_COMMIT).contains(&event_type) {
220            return Err(DecodeError::UnknownEventType(event_type));
221        }
222        let header = self.header()?;
223        let event = match event_type {
224            EVENT_GET_VERSION => Event::GetVersion(GetVersion {
225                header,
226                latency: self.f64()?,
227                priority: self.u32()?,
228                read_version: self.i64()?,
229            }),
230            EVENT_GET => Event::Get(Get {
231                header,
232                latency: self.f64()?,
233                value_size: self.i32()?,
234                key: self.bytes()?,
235            }),
236            EVENT_GET_RANGE => Event::GetRange(GetRange {
237                header,
238                latency: self.f64()?,
239                range_size: self.i32()?,
240                range: self.key_range()?,
241            }),
242            EVENT_COMMIT => Event::Commit(Commit {
243                header,
244                latency: self.f64()?,
245                num_mutations: self.i32()?,
246                commit_bytes: self.i32()?,
247                commit_version: self.i64()?,
248                request: self.commit_request(version)?,
249            }),
250            EVENT_ERROR_GET => Event::GetError(GetError {
251                header,
252                error_code: self.i32()?,
253                key: self.bytes()?,
254            }),
255            EVENT_ERROR_GET_RANGE => Event::GetRangeError(GetRangeError {
256                header,
257                error_code: self.i32()?,
258                range: self.key_range()?,
259            }),
260            EVENT_ERROR_COMMIT => Event::CommitError(CommitError {
261                header,
262                error_code: self.i32()?,
263                request: self.commit_request(version)?,
264            }),
265            other => return Err(DecodeError::UnknownEventType(other)),
266        };
267        Ok(event)
268    }
269
270    /// A serialized `KeyRangeRef`. Mirrors its deserializer: the single-key range
271    /// `[k, k\x00)` is written as `(k\x00, "")` and expanded back here.
272    fn conflict_range(&mut self) -> Result<KeyRange, DecodeError> {
273        let offset = self.pos;
274        let mut range = self.key_range()?;
275        if range.end.is_empty() && !range.begin.is_empty() {
276            if range.begin.last() != Some(&0) {
277                return Err(DecodeError::MalformedKeyRange { offset });
278            }
279            range.end = range.begin.clone();
280            range.begin.pop();
281        }
282        Ok(range)
283    }
284
285    fn commit_request(&mut self, version: ProtocolVersion) -> Result<CommitRequest, DecodeError> {
286        let read_conflict_ranges = self.vec(MIN_KEY_RANGE_LEN, Self::conflict_range)?;
287        let write_conflict_ranges = self.vec(MIN_KEY_RANGE_LEN, Self::conflict_range)?;
288        let mutations = self.vec(MIN_MUTATION_LEN, Self::mutation)?;
289        let read_snapshot = self.i64()?;
290        let report_conflicting_keys = self.bool()?;
291        let lock_aware = self.bool()?;
292        let span_context = if self.bool()? {
293            Some(self.span_context(version)?)
294        } else {
295            None
296        };
297        Ok(CommitRequest {
298            read_conflict_ranges,
299            write_conflict_ranges,
300            mutations,
301            read_snapshot,
302            report_conflicting_keys,
303            lock_aware,
304            span_context,
305        })
306    }
307
308    fn span_context(&mut self, version: ProtocolVersion) -> Result<SpanContext, DecodeError> {
309        let trace_id = [self.u64()?, self.u64()?];
310        if has_otel_span_context(version) {
311            Ok(SpanContext {
312                trace_id,
313                span_id: self.u64()?,
314                flags: self.u8()?,
315            })
316        } else {
317            Ok(SpanContext {
318                trace_id,
319                span_id: 0,
320                flags: 0,
321            })
322        }
323    }
324
325    /// Mirrors the deserializing branch of `MutationRef::serialize`.
326    fn mutation(&mut self) -> Result<Mutation, DecodeError> {
327        let offset = self.pos;
328        let mut mutation_type = self.u8()?;
329        let mut param1 = self.bytes()?;
330        let mut param2 = self.bytes()?;
331        let malformed = DecodeError::MalformedMutation { offset };
332
333        if mutation_type & CHECKSUM_FLAG_MASK != 0 {
334            let mut suffix = CHECKSUM_LEN;
335            if mutation_type & ACCUMULATIVE_CHECKSUM_INDEX_FLAG_MASK != 0 {
336                suffix += ACCUMULATIVE_CHECKSUM_INDEX_LEN;
337            }
338            let len = param2.len().checked_sub(suffix).ok_or(malformed)?;
339            param2.truncate(len);
340            mutation_type &= !(CHECKSUM_FLAG_MASK | ACCUMULATIVE_CHECKSUM_INDEX_FLAG_MASK);
341        }
342
343        // A single-key clear `[k, k\x00)` is written as `(ClearRange, k\x00, "")`. Like the
344        // C++ deserializer, a `param1` not ending in `\x00` is only traced as an error, not
345        // treated as fatal: the mutation is still expanded, dropping its last byte.
346        if mutation_type == Mutation::CLEAR_RANGE && param2.is_empty() && !param1.is_empty() {
347            if param1.last() != Some(&0) {
348                tracing::warn!(
349                    offset,
350                    param1 = ?param1,
351                    "single-key clear range mutation with param1 not ending in \\x00"
352                );
353            }
354            param2 = param1.clone();
355            param1.pop();
356        }
357
358        Ok(Mutation {
359            mutation_type,
360            param1,
361            param2,
362        })
363    }

619    fn commit_request_of(blob: &[u8]) -> Result<CommitRequest, DecodeError> {
620        match decode_events(blob)?.1.pop() {
621            Some(Event::Commit(c)) => Ok(c.request),
622            other => panic!("unexpected {other:?}"),
623        }

803    fn random_garbage_never_panics() {
804        // xorshift64, deterministic
805        let mut state = 0x9E37_79B9_7F4A_7C15u64;
806        let mut next = move || {
807            state ^= state << 13;
808            state ^= state >> 7;
809            state ^= state << 17;
810            state
811        };
812        let valid = all_events_blob();
813        for i in 0..20_000 {
814            let blob = if i % 2 == 0 {
815                // pure garbage after a valid version
816                let mut blob = ProtocolVersion::V7_4.0.to_le_bytes().to_vec();
817                let len = (next() % 256) as usize;
818                blob.extend((0..len).map(|_| next() as u8));
819                blob
820            } else {
821                // a valid blob with a few corrupted bytes
822                let mut blob = valid.clone();
823                for _ in 0..(1 + next() % 4) {
824                    let pos = 8 + (next() as usize) % (blob.len() - 8);
825                    blob[pos] = next() as u8;
826                }
827                blob
828            };
829            let _ = decode_events(&blob);
830        }
831    }
