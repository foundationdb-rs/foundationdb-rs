1//! Paged, resumable reading of the client profiling keyspace.
2//!
3//! Every sampled transaction is stored as one or more chunks under
4//! `\xff\x02/fdbClientInfo/client_latency/`. A chunk key is laid out as:
5//!
6//! ```text
7//! PREFIX | versionstamp (10) | '/' | transaction id (16) | '/' | chunk (4, BE) | total (4, BE) | '/' | user id ...
8//! ```
9//!
10//! [`ProfileScanner::read_page`] consumes the rows of [`ProfileScanner::range`], read by
11//! the caller, reassembles the chunks of each transaction, decodes them with
12//! [`decode_events`] and returns a [`Page`] with a [`Cursor`] to resume from.
13

22/// Prefix of the client profiling keyspace.
23///
24/// Defined by the [C++ client](https://github.com/apple/foundationdb/blob/main/contrib/transaction_profiling_analyzer/transaction_profiling_analyzer.py#L421).
25pub const PROFILE_PREFIX: &[u8; 32] = b"\xff\x02/fdbClientInfo/client_latency/";
26
27/// Exclusive end of the client profiling keyspace (`strinc(PROFILE_PREFIX)`).
28const PROFILE_END: &[u8; 32] = b"\xff\x02/fdbClientInfo/client_latency0";
29
30const VERSIONSTAMP_LEN: usize = 10;
31const ID_LEN: usize = 16;
32const VERSIONSTAMP_START: usize = PROFILE_PREFIX.len();
33const VERSIONSTAMP_END: usize = VERSIONSTAMP_START + VERSIONSTAMP_LEN;
34const ID_START: usize = VERSIONSTAMP_END + 1;
35const ID_END: usize = ID_START + ID_LEN;
36const CHUNK_START: usize = ID_END + 1;
37const TOTAL_START: usize = CHUNK_START + 4;
38/// Shortest chunk key the [`Assembler`] can parse: everything up to the total chunk
39/// count.
40const MIN_KEY_LEN: usize = TOTAL_START + 4;

42/// Opaque, resumable and persistable position in the profiling keyspace: the key the next
43/// page starts reading from.
44///
45/// Persist it with [`Cursor::as_bytes`] and restore it with [`Cursor::from_bytes`] to
46/// resume reading later, for instance to tail the keyspace.
47#[derive(Debug, Clone, PartialEq, Eq, Hash)]
48pub struct Cursor {
49    /// Inclusive begin key of the next read, starts with [`PROFILE_PREFIX`].
50    key: Vec<u8>,
51}
52
53/// Bytes given to [`Cursor::from_bytes`] are not a serialized [`Cursor`].
54#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
55#[error("invalid profiling cursor")]
56pub struct InvalidCursor;
57
58impl Cursor {
59    /// Cursor at the beginning of the profiling keyspace.
60    #[instrument(level = "trace")]
61    pub fn beginning() -> Self {
62        Cursor {
63            key: PROFILE_PREFIX.to_vec(),
64        }
65    }
66
67    /// Cursor at the first record with a commit version greater than or equal to
68    /// `version`.
69    ///
70    /// The version is the one at which the profiling record was written by the client,
71    /// which is some time (up to the client's flush interval) after the profiled
72    /// transaction ran. Negative versions are treated as 0.
73    #[instrument(level = "trace")]
74    pub fn at_version(version: i64) -> Self {
75        Cursor {
76            key: version_key(version),
77        }
78    }
79
80    /// Serialized form of the cursor, to persist it.
81    #[instrument(level = "trace", skip_all)]
82    pub fn as_bytes(&self) -> &[u8] {
83        &self.key
84    }
85
86    /// Restores a cursor persisted with [`Cursor::as_bytes`].
87    ///
88    /// # Errors
89    ///
90    /// Returns [`InvalidCursor`] when `bytes` is not a serialized cursor.
91    #[instrument(level = "trace", skip_all, fields(len = bytes.len()))]
92    pub fn from_bytes(bytes: Vec<u8>) -> Result<Self, InvalidCursor> {
93        if bytes.starts_with(PROFILE_PREFIX) {
94            Ok(Cursor { key: bytes })
95        } else {
96            Err(InvalidCursor)
97        }
98    }
99
100    /// Inclusive begin key of the next read.
101    fn key(&self) -> &[u8] {
102        &self.key
103    }
104}

106/// `PROFILE_PREFIX | version (8, BE) | \x00\x00`: the first key of that version.
107fn version_key(version: i64) -> Vec<u8> {
108    let mut key = PROFILE_PREFIX.to_vec();
109    key.extend_from_slice(&version.max(0).to_be_bytes());
110    key.extend_from_slice(b"\x00\x00");
111    key
112}
113
114/// The key right after `key`.
115fn key_after(key: &[u8]) -> Vec<u8> {
116    [key, b"\x00"].concat()
117}

128/// The key range the caller must read for one page, see [`ProfileScanner::range`].
129#[derive(Debug, Clone, PartialEq, Eq)]
130pub struct ScanRange {
131    /// Inclusive begin key.
132    pub begin: Vec<u8>,
133    /// Exclusive end key.
134    pub end: Vec<u8>,
135}

137/// Error of [`ProfileScanner::read_page`].
138#[derive(Debug, thiserror::Error)]
139#[non_exhaustive]
140pub enum ScanError<E> {
141    /// The row stream failed. Map it back to the caller's error, for instance to the
142    /// `FdbError` of a range read so that the caller's retry loop handles it.
143    #[error(transparent)]
144    Source(E),
145    /// The stream yielded a row outside [`ProfileScanner::range`], or a key not strictly
146    /// greater than the previous one: the caller did not read the range as required.
147    #[error("profiling row {key:?} is outside the scanned range or not after the previous row")]
148    InvalidRows {
149        /// Key of the offending row.
150        key: Vec<u8>,
151    },
152}
153
154impl<E> ScanError<E> {
155    /// Converts this error into the caller's error type: a stream error through `From`,
156    /// any other error boxed and passed to `custom`.
157    ///
158    /// The boxed error does not depend on `E`: it is a [`ScanError<std::convert::Infallible>`]
159    /// carrying the same variant data.
160    ///
161    /// With fdb-rs: `.map_err(|e| e.into_error(FdbBindingError::new_custom_error))?`, so a
162    /// failed range read reaches `db.run` as the original `FdbError` and is retried.
163    #[instrument(level = "trace", skip_all)]
164    pub fn into_error<T: From<E>>(
165        self,
166        custom: impl FnOnce(Box<dyn std::error::Error + Send + Sync>) -> T,
167    ) -> T {
168        match self {
169            ScanError::Source(err) => T::from(err),
170            ScanError::InvalidRows { key } => {
171                custom(Box::new(
172                    ScanError::<std::convert::Infallible>::InvalidRows { key },
173                ))
174            }
175        }
176    }
177}

179impl ProfileScanner {
180    /// A scanner with no [`end_version`](Self::end_version).
181    #[must_use]
182    #[instrument(level = "trace")]
183    pub fn new() -> Self {
184        Self::default()
185    }
186
187    /// Exclusive upper bound on the record commit version read by a page. Unset (the
188    /// default) reads to the end of the keyspace.
189    #[must_use]
190    #[instrument(level = "trace", skip_all)]
191    pub fn end_version(self, version: i64) -> Self {
192        ProfileScanner {
193            end_version: Some(version),
194        }
195    }
196
197    /// The key range the caller must read for the page starting at `cursor`, and give to
198    /// [`read_page`](Self::read_page): forward, with snapshot reads and no row limit,
199    /// from `begin` (inclusive) to `end` (exclusive), both as `first_greater_or_equal`
200    /// key selectors. The range is empty when `cursor` is at or past
201    /// [`end_version`](Self::end_version).
202    #[instrument(level = "trace", skip_all)]
203    pub fn range(&self, cursor: &Cursor) -> ScanRange {
204        let begin = cursor.key().to_vec();
205        let end = match self.end_version {
206            Some(version) => version_key(version),
207            None => PROFILE_END.to_vec(),
208        };
209        // never an inverted range
210        let end = end.max(begin.clone());
211        ScanRange { begin, end }
212    }
213}

256/// Result of [`ProfileScanner::read_page`].
257#[derive(Debug, Clone, PartialEq)]
258#[non_exhaustive]
259pub struct Page {
260    /// Decoded transactions, in the order they completed (the key of their last chunk).
261    pub transactions: Vec<ProfiledTransaction>,

268    /// the page.

270}

272impl ProfileScanner {
273    /// Builds one page of profiled transactions from `rows`, the rows of
274    /// [`range(cursor)`](Self::range) read by the caller, stopping when `should_stop`
275    /// says so.
276    ///
277    /// `rows` must yield exactly the key-value pairs of [`range(cursor)`](Self::range),
278    /// read forward, with snapshot reads and no row limit: every row inside the range,
279    /// in strictly increasing key order. A row outside the range or out of order fails
280    /// the page with [`ScanError::InvalidRows`]; an error of the stream fails it with
281    /// [`ScanError::Source`]. The stream ending means the end of the range was reached,
282    /// and the page is [`exhausted`](Page::exhausted).
283    ///
284    /// `should_stop` is where the caller bounds the page (a transaction budget, a row
285    /// count, anything): it is called after every row, and the page stops as soon as it
286    /// returns `true`, without polling `rows` again. Then:
287    ///
288    /// - With no transaction pending (all the chunks read so far belong to complete or
289    ///   broken transactions), [`Page::next`] points right after the last row.
290    /// - Otherwise [`Page::next`] points at the first chunk of the oldest pending
291    ///   transaction, before every transaction the page completed after it, which the
292    ///   page drops and the next one reads again. Such a transaction is returned whole
293    ///   by a later page, even when its chunks were written in two commits with other
294    ///   rows in between.
295    /// - When that position is `cursor` itself (a single transaction cannot be read from
296    ///   `cursor` before the caller says stop), the pending transactions are reported as
297    ///   [`SkipReason::BrokenChunks`] and [`Page::next`] points right after the last
298    ///   row, so every stopped page advances the cursor.
299    ///
300    /// Each transaction is returned once over a sequence of pages that each resume from
301    /// the previous [`Page::next`]; one is lost only in the last case above. The client
302    /// may write the chunks of one transaction in two consecutive commits (when a flush
303    /// exceeds the transaction size limit), with rows of other transactions in between.
304    /// When the range ends while such a transaction is still incomplete (its second
305    /// commit is not written yet, or lies beyond [`end_version`](Self::end_version)),
306    /// [`Page::next`] also points at its first chunk: a bounded read never returns such
307    /// a transaction, an unbounded read from that cursor does once it is complete.
308    ///
309    /// A chunk whose earlier chunks are missing is reported as
310    /// [`SkipReason::BrokenChunks`], once per transaction id and page, including chunks
311    /// of a transaction that began before the cursor the scan started from
312    /// ([`Cursor::beginning`], [`Cursor::at_version`]). A transaction that fails to
313    /// decode is reported in [`Page::skipped`] and does not fail the page.
314    ///
315    /// # Errors
316    ///
317    /// [`ScanError::Source`] with the stream's error, [`ScanError::InvalidRows`] when the
318    /// rows do not follow the contract above.
319    #[instrument(
320        level = "debug",
321        skip_all,
322        fields(
323            end_version = ?self.end_version,
324            rows,
325            transactions,
326            skipped,
327            exhausted,
328        )
329    )]
330    pub async fn read_page<S, K, V, E>(
331        &self,
332        cursor: &Cursor,
333        rows: S,
334        mut should_stop: impl FnMut() -> bool,
335    ) -> Result<Page, ScanError<E>>
336    where
337        S: Stream<Item = Result<(K, V), E>>,
338        K: AsRef<[u8]>,
339        V: AsRef<[u8]>,
340    {
341        let range = self.range(cursor);
342        let mut assembler = Assembler::new(cursor);
343        let mut rows = pin!(rows);
344        let mut read = 0usize;
345        let page = loop {
346            let Some(row) = poll_fn(|cx| rows.as_mut().poll_next(cx)).await else {
347                break assembler.finish(false);
348            };
349            let (key, value) = row.map_err(ScanError::Source)?;
350            let key = key.as_ref();
351            let in_order = assembler.last_key.as_deref().is_none_or(|last| key > last);
352            if key < range.begin.as_slice() || key >= range.end.as_slice() || !in_order {
353                return Err(ScanError::InvalidRows { key: key.to_vec() });
354            }
355            assembler.push(key, value.as_ref());
356            read += 1;
357            if should_stop() {
358                break assembler.finish(true);
359            }
360        };
361
362        let span = tracing::Span::current();
363        span.record("rows", read);
364        span.record("transactions", page.transactions.len());
365        span.record("skipped", page.skipped.len());
366        span.record("exhausted", page.exhausted);
367        Ok(page)
368    }
369}

371fn version_of(versionstamp: &[u8; 10]) -> i64 {
372    let mut version = [0u8; 8];
373    version.copy_from_slice(&versionstamp[..8]);
374    i64::from_be_bytes(version)
375}

377/// The fields of a chunk key.
378#[derive(Debug, Clone, Copy, PartialEq, Eq)]
379struct ChunkKey {
380    versionstamp: [u8; 10],
381    id: [u8; 16],
382    chunk: u32,
383    total: u32,
384}
385
386impl ChunkKey {
387    /// Parses a chunk key by offsets, ignoring what follows the total chunk count.
388    /// Returns `None` for a key that is too short or has misplaced separators.
389    fn parse(key: &[u8]) -> Option<Self> {
390        if key.len() < MIN_KEY_LEN
391            || !key.starts_with(PROFILE_PREFIX)
392            || key[VERSIONSTAMP_END] != b'/'
393            || key[ID_END] != b'/'
394        {
395            return None;
396        }
397        let mut versionstamp = [0u8; VERSIONSTAMP_LEN];
398        versionstamp.copy_from_slice(&key[VERSIONSTAMP_START..VERSIONSTAMP_END]);
399        let mut id = [0u8; ID_LEN];
400        id.copy_from_slice(&key[ID_START..ID_END]);
401        let be = |start: usize| {
402            let mut n = [0u8; 4];
403            n.copy_from_slice(&key[start..start + 4]);
404            u32::from_be_bytes(n)
405        };
406        Some(ChunkKey {
407            versionstamp,
408            id,
409            chunk: be(CHUNK_START),
410            total: be(TOTAL_START),
411        })
412    }
413}

415/// A reassembly outcome of the [`Assembler`].
416#[derive(Debug, Clone, PartialEq, Eq)]
417enum Assembled {
418    /// Every chunk of a transaction, concatenated.
419    Complete {
420        versionstamp: [u8; 10],
421        id: [u8; 16],
422        blob: Vec<u8>,
423    },

425    Broken {

431/// An [`Assembled`] outcome with the keys of the rows it was made of.
432#[derive(Debug)]
433struct Output {

436    /// Key of the last row of the transaction read by this page: the one that produced

442/// A multi-chunk transaction being reassembled.
443#[derive(Debug)]
444struct Partial {
445    /// Key of its first chunk.

447    id: [u8; 16],
448    /// Versionstamp of its first chunk.
449    versionstamp: [u8; 10],

459/// Builds one [`Page`] from the rows of its range, fed in key order. Pure, no I/O:
460/// [`ProfileScanner::read_page`] feeds it the caller's rows and tells it where it stopped.
461///
462/// A multi-chunk transaction starts at chunk 1 and its chunks come in order 1..=total,
463/// with the same id and total and non-decreasing versionstamps: the client may write them
464/// in two commits, so rows of other transactions can sit in between. Anything else
465/// drops the partial buffer and reports the transaction as [`Assembled::Broken`], once
466/// per id until a new chunk 1 of that id.
467#[derive(Debug)]
468struct Assembler {
469    /// Where the page started.
470    start: Cursor,
471    /// Pending transactions by id.
472    partials: HashMap<[u8; 16], Partial>,
473    /// Ids reported as broken, with the index of their report in `outputs`, so that their
474    /// remaining chunks are consumed silently. A new chunk 1 of the id removes it.
475    broken: HashMap<[u8; 16], usize>,
476    /// Outcomes, in the order they were produced.
477    outputs: Vec<Output>,
478    /// Last row fed.
479    last_key: Option<Vec<u8>>,
480}
481
482impl Assembler {
483    fn new(start: &Cursor) -> Self {
484        Assembler {
485            start: start.clone(),
486            partials: HashMap::new(),
487            broken: HashMap::new(),
488            outputs: Vec::new(),
489            last_key: None,
490        }
491    }
492
493    /// Feeds one row.
494    fn push(&mut self, key: &[u8], value: &[u8]) {
495        self.last_key = Some(key.to_vec());
496        match ChunkKey::parse(key) {
497            Some(parsed) => self.process(parsed, key, value),
498            None => tracing::warn!(key = ?key, "ignoring unparseable profiling key"),
499        }
500    }
501
502    /// Processes a parsed row.
503    fn process(&mut self, parsed: ChunkKey, key: &[u8], value: &[u8]) {
504        if let Some(mut partial) = self.partials.remove(&parsed.id) {
505            if parsed.total == partial.total
506                && parsed.chunk == partial.next_chunk
507                && parsed.versionstamp >= partial.last_versionstamp
508            {
509                partial.blob.extend_from_slice(value);
510                if parsed.chunk == partial.total {
511                    self.outputs.push(Output {
512                        first_key: partial.first_key,
513                        last_key: key.to_vec(),
514                        assembled: Assembled::Complete {
515                            versionstamp: partial.versionstamp,
516                            id: parsed.id,
517                            blob: partial.blob,
518                        },
519                    });
520                    return;
521                }
522                // chunk < total, so this cannot overflow
523                partial.next_chunk += 1;
524                partial.last_versionstamp = parsed.versionstamp;
525                self.partials.insert(parsed.id, partial);
526                return;
527            }
528            self.report_broken(partial.first_key, partial.versionstamp, parsed.id, key);
529        }
530
531        if parsed.chunk == 1 && parsed.total >= 1 {
532            // a new transaction under this id, its chunks are no longer those of a broken one
533            self.broken.remove(&parsed.id);
534        }
535        if parsed.chunk == 1 && parsed.total == 1 {
536            self.outputs.push(Output {
537                first_key: key.to_vec(),
538                last_key: key.to_vec(),
539                assembled: Assembled::Complete {
540                    versionstamp: parsed.versionstamp,
541                    id: parsed.id,
542                    blob: value.to_vec(),
543                },
544            });
545            return;
546        }
547        if parsed.chunk == 1 && parsed.total > 1 {
548            self.partials.insert(
549                parsed.id,
550                Partial {
551                    first_key: key.to_vec(),
552                    id: parsed.id,
553                    versionstamp: parsed.versionstamp,
554                    last_versionstamp: parsed.versionstamp,
555                    total: parsed.total,
556                    next_chunk: 2,
557                    blob: value.to_vec(),
558                },
559            );
560        } else {
561            self.report_broken(key.to_vec(), parsed.versionstamp, parsed.id, key);
562        }
563    }
564
565    /// Reports `id` as broken, unless it already is: then `key` becomes the last row of
566    /// that report.
567    fn report_broken(
568        &mut self,
569        first_key: Vec<u8>,
570        versionstamp: [u8; 10],
571        id: [u8; 16],
572        key: &[u8],
573    ) {
574        if let Some(&index) = self.broken.get(&id) {
575            if let Some(output) = self.outputs.get_mut(index) {
576                output.last_key = key.to_vec();
577            }
578            return;
579        }
580        self.broken.insert(id, self.outputs.len());
581        self.outputs.push(Output {
582            first_key,
583            last_key: key.to_vec(),
584            assembled: Assembled::Broken { versionstamp, id },
585        });
586    }
587
588    /// Where the next page resumes to read again the transaction whose first chunk is at
589    /// `cut`: lowers `cut` to the first key of every outcome whose last row is at or after
590    /// it, up to a fixed point. Returns the cut and the indexes of those outcomes, which
591    /// the next page reports instead.
592    fn cut(&self, mut cut: Vec<u8>) -> (Vec<u8>, Vec<usize>) {
593        let mut by_last: Vec<usize> = (0..self.outputs.len()).collect();
594        by_last.sort_by(|&a, &b| self.outputs[b].last_key.cmp(&self.outputs[a].last_key));
595        let mut dropped = Vec::new();
596        for index in by_last {
597            let output = &self.outputs[index];
598            if output.last_key < cut {
599                break;
600            }
601            dropped.push(index);
602            if output.first_key < cut {
603                cut = output.first_key.clone();
604            }
605        }
606        (cut, dropped)
607    }
608
609    /// Builds the page. `stopped` tells whether the caller's check stopped it right after
610    /// its last row, rather than the end of its range.
611    ///
612    /// Pending transactions are left to the next page: it resumes at the
613    /// [`cut`](Self::cut) before them and reads them again. When the page was stopped and
614    /// that cut would not advance past the cursor, it resumes after the last row and
615    /// breaks them instead. With nothing pending, it resumes after the last row.
616    fn finish(mut self, stopped: bool) -> Page {
617        let mut pending: Vec<Partial> = std::mem::take(&mut self.partials).into_values().collect();
618        pending.sort_by(|a, b| a.first_key.cmp(&b.first_key));
619        let after_last = match &self.last_key {
620            Some(last) => key_after(last),
621            None => self.start.key().to_vec(),
622        };
623
624        let mut kept = vec![true; self.outputs.len()];
625        let cut = match pending.first() {
626            None => None,
627            Some(oldest) => {
628                let (cut, dropped) = self.cut(oldest.first_key.clone());
629                if stopped && cut.as_slice() <= self.start.key() {
630                    // a cut at the cursor would not progress: break the pending ones
631                    None
632                } else {
633                    for &index in &dropped {
634                        kept[index] = false;
635                    }
636                    tracing::debug!(
637                        pending = pending.len(),
638                        left_to_next_page = dropped.len(),
639                        stopped,
640                        "page ended with incomplete transactions"
641                    );
642                    Some(cut)
643                }
644            }
645        };
646        let next = match cut {
647            Some(cut) => cut,
648            None => {
649                // the page resumes after its last row: what is still pending cannot be
650                // reassembled (with no cut, only a stopped page has pending ones)
651                for partial in pending {
652                    tracing::debug!("breaking incomplete transaction at a stop");
653                    self.outputs.push(Output {
654                        last_key: partial.first_key.clone(),
655                        first_key: partial.first_key,
656                        assembled: Assembled::Broken {
657                            versionstamp: partial.versionstamp,
658                            id: partial.id,
659                        },
660                    });
661                    kept.push(true);
662                }
663                after_last
664            }
665        };
666
667        let mut transactions = Vec::new();
668        let mut skipped = Vec::new();
669        for (output, kept) in self.outputs.into_iter().zip(kept) {
670            if !kept {
671                continue;
672            }
673            match output.assembled {
674                Assembled::Complete {
675                    versionstamp,
676                    id,
677                    blob,
678                } => match decode_events(&blob) {
679                    Ok((protocol_version, events)) => transactions.push(ProfiledTransaction {
680                        version: version_of(&versionstamp),
681                        versionstamp,
682                        id,
683                        protocol_version,
684                        events,
685                    }),
686                    Err(err) => {
687                        tracing::debug!(error = %err, "skipping undecodable transaction");
688                        skipped.push(Skipped {
689                            versionstamp,
690                            id,
691                            reason: SkipReason::Decode(err),
692                        })
693                    }
694                },
695                Assembled::Broken { versionstamp, id } => {
696                    tracing::debug!("skipping transaction with broken chunks");
697                    skipped.push(Skipped {
698                        versionstamp,
699                        id,
700                        reason: SkipReason::BrokenChunks,
701                    })
702                }
703            }
704        }
705
706        Page {
707            transactions,
708            skipped,
709            next: Cursor { key: next },
710            exhausted: !stopped,
711        }
712    }
713}

817    /// Runs `read_page` over `rows`, which must be ready (no I/O in these tests).
818    fn read<E>(
819        scanner: &ProfileScanner,
820        cursor: &Cursor,
821        rows: Vec<Result<Row, E>>,
822        should_stop: impl FnMut() -> bool,
823    ) -> Result<Page, ScanError<E>> {
824        scanner

826            .now_or_never()
827            .expect("a ready stream completes the page in one poll")

830    /// Runs one page over the sorted `rows`, feeding the ones of its range like a caller
831    /// honoring the contract of `read_page`.
832    fn run_page(rows: &Rows, cursor: &Cursor, should_stop: impl FnMut() -> bool) -> Page {
833        let scanner = ProfileScanner::new();
834        let range = scanner.range(cursor);
835        let rows = rows
836            .iter()
837            .filter(|(k, _)| *k >= range.begin && *k < range.end)
838            .cloned()
839            .map(Ok::<_, Infallible>)
840            .collect();
841        match read(&scanner, cursor, rows, should_stop) {
842            Ok(page) => page,
843            Err(err) => panic!("{err}"),
844        }
845    }

847    /// Runs pages until exhausted, each with a fresh stop check, returns every reported
848    /// transaction and skip.
849    fn run_pages<C: FnMut() -> bool>(
850        rows: &Rows,
851        mut should_stop: impl FnMut() -> C,
852    ) -> (Vec<ProfiledTransaction>, Vec<Skipped>) {
853        let mut cursor = Cursor::beginning();
854        let (mut txs, mut skipped) = (Vec::new(), Vec::new());
855        for _ in 0..10_000 {
856            let page = run_page(rows, &cursor, should_stop());
857            txs.extend(page.transactions);
858            skipped.extend(page.skipped);
859            let next = Cursor::from_bytes(page.next.as_bytes().to_vec()).unwrap();
860            assert_eq!(next, page.next);
861            if page.exhausted {
862                return (txs, skipped);
863            }
864            assert!(next.key() > cursor.key(), "no progress");
865            cursor = next;
866        }
867        panic!("pages never exhausted");
868    }
