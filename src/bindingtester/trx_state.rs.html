1//! The state of a named transaction of the stack machine.
2
3use foundationdb as fdb;
4
5use std::mem::transmute;
6
7use fdb::*;
8
9pub(crate) enum TransactionState {
10    Transaction(Transaction),
11    TransactionCancelled(TransactionCancelled),
12    TransactionCommitted(TransactionCommitted),
13    TransactionCommitError(TransactionCommitError),
14    Pending(usize),
15    Dead,
16}
17
18impl std::fmt::Debug for TransactionState {
19    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
20        use TransactionState as S;
21
22        match self {
23            S::Transaction(..) => "Transaction",
24            S::TransactionCancelled(..) => "TransactionCancelled",
25            S::TransactionCommitted(..) => "TransactionCommitted",
26            S::TransactionCommitError(..) => "TransactionCommitError",
27            S::Pending(..) => "Pending",
28            S::Dead => "Dead",
29        }
30        .fmt(f)
31    }
32}
33
34impl TransactionState {
35    pub(crate) fn reset(&mut self) {
36        use TransactionState as S;
37        *self = match std::mem::replace(self, S::Dead) {
38            S::TransactionCommitted(c) => S::Transaction(c.reset()),
39            S::TransactionCommitError(c) => S::Transaction(c.reset()),
40            c => c,
41        }
42    }
43
44    pub(crate) fn as_mut(&mut self) -> &mut Transaction {
45        use TransactionState as S;
46
47        self.reset();
48        match *self {
49            S::Transaction(ref mut tr) => tr,
50            S::TransactionCancelled(ref mut tr) => unsafe {
51                // rust binding prevent accessing cancelled transaction
52                &mut *(tr as *mut TransactionCancelled as *mut Transaction)
53            },
54            _ => panic!("transaction is owned by a future that is still not done"),
55        }
56    }
57
58    pub(crate) fn take(&mut self, id: usize) -> Transaction {
59        use TransactionState as S;
60
61        self.reset();
62        match std::mem::replace(self, S::Dead) {
63            S::Transaction(tr) => {
64                *self = S::Pending(id);
65                tr
66            }
67            S::TransactionCancelled(tr) => {
68                *self = S::Pending(id);
69                unsafe {
70                    // rust binding prevent accessing cancelled transaction
71                    transmute::<TransactionCancelled, Transaction>(tr)
72                }
73            }
74            _ => panic!("transaction is owned by a future that is still not done"),
75        }
76    }
77}
