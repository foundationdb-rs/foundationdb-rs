10pub(crate) fn mutation_from_str(s: &str) -> MutationType {
11    match s {
12        "ADD" => MutationType::Add,
13        "AND" => MutationType::And,
14        "BIT_AND" => MutationType::BitAnd,
15        "OR" => MutationType::Or,
16        "BIT_OR" => MutationType::BitOr,
17        "XOR" => MutationType::Xor,
18        "BIT_XOR" => MutationType::BitXor,
19        "APPEND_IF_FITS" => MutationType::AppendIfFits,
20        "MAX" => MutationType::Max,
21        "MIN" => MutationType::Min,
22        "SET_VERSIONSTAMPED_KEY" => MutationType::SetVersionstampedKey,
23        "SET_VERSIONSTAMPED_VALUE" => MutationType::SetVersionstampedValue,
24        "BYTE_MIN" => MutationType::ByteMin,
25        "BYTE_MAX" => MutationType::ByteMax,
26        "COMPARE_AND_CLEAR" => MutationType::CompareAndClear,
27        _ => unimplemented!("mutation_from_str({})", s),
28    }
29}

31pub(crate) fn streaming_from_value(val: i32) -> StreamingMode {
32    match val {
33        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_WANT_ALL => StreamingMode::WantAll,
34        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_ITERATOR => StreamingMode::Iterator,
35        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_EXACT => StreamingMode::Exact,

37        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_MEDIUM => StreamingMode::Medium,
38        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_LARGE => StreamingMode::Large,
39        fdb_sys::FDBStreamingMode_FDB_STREAMING_MODE_SERIAL => StreamingMode::Serial,
40        _ => unimplemented!("streaming_from_value({})", val),

52impl std::fmt::Debug for Instr {
53    fn fmt(&self, fmt: &mut std::fmt::Formatter) -> std::fmt::Result {
54        write!(fmt, "[{:?}", self.code)?;
55        if self.database {
56            write!(fmt, " db")?;
57        }
58        if self.snapshot {
59            write!(fmt, " snapshot")?;
60        }
61        if self.starts_with {
62            write!(fmt, " starts_with")?;
63        }
64        if self.selector {
65            write!(fmt, " selector")?;
66        }
67        write!(fmt, "]")
68    }
69}

71impl Instr {
72    pub(crate) fn pop_database(&mut self) -> bool {
73        if self.database {
74            self.database = false;
75            true
76        } else {
77            false
78        }
79    }
80    pub(crate) fn pop_snapshot(&mut self) -> bool {
81        if self.snapshot {
82            self.snapshot = false;
83            true
84        } else {
85            false
86        }
87    }
88    pub(crate) fn pop_starts_with(&mut self) -> bool {
89        if self.starts_with {
90            self.starts_with = false;
91            true
92        } else {
93            false
94        }
95    }
96    pub(crate) fn pop_selector(&mut self) -> bool {
97        if self.selector {
98            self.selector = false;
99            true
100        } else {
101            false
102        }
103    }
104
105    pub(crate) fn has_flags(&self) -> bool {
106        self.database || self.snapshot || self.starts_with || self.selector
107    }

201pub(crate) fn has_opt<'a>(cmd: &'a str, opt: &'static str) -> (&'a str, bool) {
202    if cmd.ends_with(opt) {
203        (&cmd[0..(cmd.len() - opt.len())], true)
204    } else {
205        (cmd, false)
206    }
207}

209impl Instr {
210    pub(crate) fn from(data: &[u8]) -> Self {
211        use crate::InstrCode::*;
212
213        let data = Bytes::from(data);
214        trace!("inst {data:?}");
215        let tup: Vec<Element> = unpack(&data).unwrap();
216        let cmd = tup[0].as_str().unwrap();
217
218        let (cmd, database) = has_opt(cmd, "_DATABASE");
219        let (cmd, snapshot) = has_opt(cmd, "_SNAPSHOT");
220        let (cmd, starts_with) = has_opt(cmd, "_STARTS_WITH");
221        let (cmd, selector) = has_opt(cmd, "_SELECTOR");
222
223        let code = match cmd {
224            "PUSH" => Push(tup[1].clone().into_owned()),
225            "DUP" => Dup,
226            "EMPTY_STACK" => EmptyStack,
227            "SWAP" => Swap,
228            "POP" => Pop,
229            "SUB" => Sub,
230            "CONCAT" => Concat,
231            "LOG_STACK" => LogStack,
232
233            "NEW_TRANSACTION" => NewTransaction,
234            "USE_TRANSACTION" => UseTransaction,
235            "ON_ERROR" => OnError,
236            "GET" => Get,
237            "GET_KEY" => GetKey,
238            "GET_RANGE" => GetRange,
239            "GET_READ_VERSION" => GetReadVersion,
240            "GET_VERSIONSTAMP" => GetVersionstamp,

242            "SET" => Set,

244            "CLEAR" => Clear,
