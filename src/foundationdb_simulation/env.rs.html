1//! Simulator-backed implementations of the [`foundationdb::env`] capabilities.
2//!
3//! A workload that reads the machine clock or draws from an ambient generator
4//! stops being reproducible, which defeats the point of running it in the
5//! simulator. [`SimClock`] and [`SimRng`] plug the simulated time and the
6//! deterministic generator of fdbserver into the [`Clock`] and [`Rng`] traits, so
7//! layer code written against an [`Environment`] replays identically from a
8//! seed.
9//!
10//! Build one from the [`WorkloadContext`] handed to your workload:
11//!
12//! ```ignore
13//! impl RustWorkload for MyWorkload {
14//!     fn new(name: String, context: WorkloadContext) -> Self {
15//!         Self {
16//!             env: context.environment(),
17//!             context,
18//!         }
19//!     }
20//! }
21//! ```
22//!
23//! # Lifetime
24//!
25//! Every type here holds a copy of the fdbserver-owned context, with the same
26//! caveat as [`WorkloadContext::clone`]: the copy aliases a context that
27//! fdbserver frees with the workload instance. Keep a [`SimClock`], a
28//! [`SimRng`], or an [`Environment`] holding them no longer than the workload
29//! that produced it. Using one afterwards dereferences a dangling pointer.
30
31use std::fmt;
32use std::sync::Arc;
33use std::time::Duration;

35use foundationdb::env::{Clock, Environment, Rng};

37use crate::bindings::WorkloadContext;

39/// A copy of the fdbserver-owned workload context.
40///
41/// The wrapped [`WorkloadContext`] is a bundle of raw pointers with no `Drop`,
42/// so copying it is free and dropping it does nothing. See the
43/// [module documentation](self) for how long the copy stays valid.
44struct ContextHandle(WorkloadContext);
45
46impl ContextHandle {
47    fn new(context: &WorkloadContext) -> Self {
48        Self(context.clone())
49    }
50}
51
52/// The [`Clock`] of a simulated workload, reading fdbserver's simulated time.
53///
54/// [`monotonic`](Clock::monotonic) is the simulated clock, which starts at zero
55/// and only moves when the simulator decides to advance it, so a run replays
56/// identically. [`wall`](Clock::wall) is that same reading: simulated wall time
57/// counts from the UNIX epoch at simulation start, second zero being the epoch.
58///
59/// Simulated wall time is therefore deterministic and, unlike a real machine
60/// clock, never jumps: a workload cannot exercise clock-skew handling this way.
61///
62/// See the [module documentation](self) for how long an instance stays valid.
63pub struct SimClock {
64    context: ContextHandle,
65}
66
67// SAFETY: the wrapped context is a bundle of raw pointers into fdbserver, which
68// drives every workload callback on a single thread, so the context is never
69// accessed concurrently. This is the same justification as the rest of the
70// crate's use of the raw context, see `Clone for WorkloadContext`.
71unsafe impl Send for SimClock {}
72// SAFETY: see the `Send` impl above.
73unsafe impl Sync for SimClock {}
74
75impl SimClock {
76    /// Reads simulated time from `context`.
77    pub fn new(context: &WorkloadContext) -> Self {
78        Self {
79            context: ContextHandle::new(context),
80        }
81    }
82}
83
84impl fmt::Debug for SimClock {
85    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
86        f.write_str("SimClock")
87    }
88}
89
90impl Clock for SimClock {
91    fn monotonic(&self) -> Duration {
92        // The simulated clock starts at zero and never goes backwards, but it is
93        // an f64 crossing the FFI boundary: clamp rather than panic.
94        Duration::from_secs_f64(self.context.0.now().max(0.0))
95    }
96
97    /// Simulated wall time counts from the UNIX epoch at simulation start, so
98    /// second zero is the epoch itself: only differences and orderings carry
99    /// meaning. It never jumps, so clock-skew handling cannot be exercised here.
100    fn wall(&self) -> Duration {
101        self.monotonic()
102    }
103}

105/// The [`Rng`] of a simulated workload, drawing from fdbserver's deterministic
106/// generator.
107///
108/// Every draw advances the simulator's own generator, so the values are part of
109/// the run's reproducible state: the same seed replays the same sequence, and
110/// drawing more or fewer values than a previous run changes what every later
111/// consumer sees.
112///
113/// A component that needs a large volume of randomness should draw one value
114/// here and seed a local generator with it, the pattern described in
115/// [`foundationdb::env`]. That keeps the simulator's sequence short and stable.
116///
117/// See the [module documentation](self) for how long an instance stays valid.
118pub struct SimRng {
119    context: ContextHandle,
120}
121
122// SAFETY: the wrapped context is a bundle of raw pointers into fdbserver, which
123// drives every workload callback on a single thread, so the context is never
124// accessed concurrently. This is the same justification as the rest of the
125// crate's use of the raw context, see `Clone for WorkloadContext`.
126unsafe impl Send for SimRng {}
127// SAFETY: see the `Send` impl above.
128unsafe impl Sync for SimRng {}
129
130impl SimRng {
131    /// Draws from the deterministic generator of `context`.
132    pub fn new(context: &WorkloadContext) -> Self {
133        Self {
134            context: ContextHandle::new(context),
135        }
136    }
137}
138
139impl fmt::Debug for SimRng {
140    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
141        f.write_str("SimRng")
142    }
143}
144
145impl Rng for SimRng {
146    /// Composes two 32-bit draws, so one call consumes two values of the
147    /// simulator's sequence.
148    fn next_u64(&self) -> u64 {
149        let high = self.context.0.rnd();
150        let low = self.context.0.rnd();
151        (u64::from(high) << 32) | u64::from(low)
152    }
153
154    /// One draw of the simulator's sequence, taken directly from the generator.
155    fn next_u32(&self) -> u32 {
156        self.context.0.rnd()
157    }
158}
159
160impl WorkloadContext {
161    /// The deterministic [`Environment`] of this workload: simulated time and
162    /// the simulator's generator.
163    ///
164    /// Pass it to the layers and recipes the workload exercises so that they
165    /// read time and randomness from the simulator instead of from the machine.
166    ///
167    /// The result borrows nothing, but it holds a copy of this context: see the
168    /// [module documentation](self) for how long it stays valid.
169    pub fn environment(&self) -> Environment {
170        Environment::new(Arc::new(SimClock::new(self)), Arc::new(SimRng::new(self)))
171    }
172}
