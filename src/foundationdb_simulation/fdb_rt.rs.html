8use std::{
9    cell::{Cell, RefCell},
10    collections::{BTreeMap, VecDeque},
11    future::Future,

13    sync::{
14        Arc, Mutex, Weak,

20type TaskId = usize;
21type Task = Pin<Box<dyn Future<Output = ()>>>;
22type Queue = Mutex<VecDeque<Notification>>;
23
24thread_local! {
25    static EXECUTOR: RefCell<Executor> = RefCell::new(Executor::default());
26    static POLLING: Cell<bool> = const { Cell::new(false) };
27}
28
29#[derive(Default)]
30struct Executor {
31    tasks: BTreeMap<TaskId, Task>,
32    queue: Arc<Queue>,
33    next_id: TaskId,
34}
35
36enum Notification {
37    Ready(Arc<TaskWaker>),
38    Cancel(TaskId),
39}
40
41/// Wakers may cross threads, but must never own or access an executor's futures.
42struct TaskWaker {
43    id: TaskId,
44    queue: Weak<Queue>,
45    queued: AtomicBool,
46}
47
48impl Wake for TaskWaker {
49    fn wake(self: Arc<Self>) {
50        if self.queued.swap(true, Ordering::AcqRel) {
51            return;
52        }
53        let Some(queue) = self.queue.upgrade() else {
54            return;
55        };
56        queue
57            .lock()
58            .unwrap_or_else(|poisoned| poisoned.into_inner())
59            .push_back(Notification::Ready(self));
60    }
61}
62
63impl Drop for TaskWaker {
64    fn drop(&mut self) {
65        // A pending future with no remaining wakers cannot make progress. Only
66        // its owner may destroy it, even when its last waker dies on another thread.
67        if let Some(queue) = self.queue.upgrade() {
68            queue
69                .lock()
70                .unwrap_or_else(|poisoned| poisoned.into_inner())
71                .push_back(Notification::Cancel(self.id));
72        }
73    }
74}
75
76struct PollingGuard;
77
78impl Drop for PollingGuard {
79    fn drop(&mut self) {
80        let _ = POLLING.try_with(|polling| polling.set(false));
81    }
82}

84/// Poll this thread's queued tasks until no notifications remain.
85///
86/// Wakes from other threads are delivered here, on the thread that spawned the
87/// future. Calling this function on a different thread cannot poll those tasks.
88/// Nested calls defer to the outer drain, so a future is never polled reentrantly.
89pub fn poll_pending_tasks() {
90    // Cancelling a future during thread teardown can invoke the callback hook.
91    // Its executor is already being destroyed and must not be entered again.
92    let Ok(queue) = EXECUTOR.try_with(|executor| executor.borrow().queue.clone()) else {
93        return;
94    };
95    let Ok(false) = POLLING.try_with(|polling| polling.replace(true)) else {
96        return;
97    };
98    let _guard = PollingGuard;
99    loop {
100        let notification = queue
101            .lock()
102            .unwrap_or_else(|poisoned| poisoned.into_inner())
103            .pop_front();
104        let Some(notification) = notification else {
105            return;
106        };
107        // Neither the queue lock nor the registry borrow may span user polling
108        // or destruction: either can synchronously wake or spawn another task.
109        match notification {
110            Notification::Ready(notification) => {
111                // A coalesced wake may publish writes without taking the queue
112                // lock. Acquire those writes before polling the future.
113                notification.queued.swap(false, Ordering::AcqRel);
114                let id = notification.id;
115                let future = EXECUTOR.with_borrow_mut(|executor| executor.tasks.remove(&id));
116                if let Some(mut future) = future {
117                    let waker = Waker::from(notification);
118                    let mut cx = Context::from_waker(&waker);
119                    if future.as_mut().poll(&mut cx).is_pending() {
120                        EXECUTOR.with_borrow_mut(|executor| {
121                            executor.tasks.insert(id, future);
122                        });
123                    }
124                }
125            }
126            Notification::Cancel(id) => {
127                let future = EXECUTOR.with_borrow_mut(|executor| executor.tasks.remove(&id));
128                drop(future);
129            }
130        }
131    }
132}

134/// Spawn an async block and resolve all contained FoundationDB futures.
135pub(crate) fn fdb_spawn<F>(future: F)
136where
137    F: Future<Output = ()> + 'static,
138{
139    let notification = EXECUTOR.with_borrow_mut(|executor| {
140        let id = executor.next_id;
141        executor.next_id = id.checked_add(1).expect("simulation task IDs exhausted");
142        executor.tasks.insert(id, Box::pin(future));
143        Arc::new(TaskWaker {
144            id,
145            queue: Arc::downgrade(&executor.queue),
146            queued: AtomicBool::new(false),
147        })
148    });
149    notification.wake();
150    poll_pending_tasks();
151}

153#[cfg(test)]
154mod tests {
155    use super::*;
156    use std::{
157        future::poll_fn,
158        rc::Rc,
159        sync::mpsc,
160        task::Poll,
161        thread::{self, ThreadId},
162    };
163
164    struct LocalFuture {
165        owner: ThreadId,
166        polls: Rc<Cell<usize>>,
167        drops: Arc<Mutex<Vec<ThreadId>>>,
168        wakers: Arc<Mutex<Vec<Waker>>>,
169        ready: Arc<AtomicBool>,
170    }
171
172    impl Future for LocalFuture {
173        type Output = ();
174
175        fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
176            assert_eq!(thread::current().id(), self.owner);
177            self.polls.set(self.polls.get() + 1);
178            if self.ready.load(Ordering::Acquire) {
179                Poll::Ready(())
180            } else {
181                self.wakers.lock().unwrap().push(cx.waker().clone());
182                Poll::Pending
183            }
184        }
185    }
186
187    impl Drop for LocalFuture {
188        fn drop(&mut self) {
189            assert_eq!(thread::current().id(), self.owner);
190            // FDB cancellation can synchronously invoke this hook, including
191            // when the owner thread is destroying its executor registry.
192            poll_pending_tasks();
193            self.drops.lock().unwrap().push(thread::current().id());
194        }
195    }
196
197    #[test]
198    fn foreign_wakes_only_poll_and_drop_the_future_on_its_owner() {
199        let owner = thread::current().id();
200        let polls = Rc::new(Cell::new(0));
201        let drops = Arc::new(Mutex::new(Vec::new()));
202        let wakers = Arc::new(Mutex::new(Vec::new()));
203        let ready = Arc::new(AtomicBool::new(false));
204        fdb_spawn(LocalFuture {
205            owner,
206            polls: polls.clone(),
207            drops: drops.clone(),
208            wakers: wakers.clone(),
209            ready: ready.clone(),
210        });
211        let waker = wakers.lock().unwrap().pop().unwrap();
212        let other_waker = waker.clone();
213        let late_waker = waker.clone();
214        ready.store(true, Ordering::Release);
215        let first = thread::spawn(move || {
216            waker.wake_by_ref();
217            waker.wake_by_ref();
218            waker.wake();
219            poll_pending_tasks();
220        });
221        let second = thread::spawn(move || {
222            other_waker.wake();
223            poll_pending_tasks();
224        });
225        first.join().unwrap();
226        second.join().unwrap();
227        assert_eq!(polls.get(), 1);
228        assert!(drops.lock().unwrap().is_empty());
229
230        poll_pending_tasks();
231        assert_eq!(polls.get(), 2);
232        assert_eq!(*drops.lock().unwrap(), [owner]);
233
234        // Retaining or waking a completed task's waker must not retain or repoll
235        // the future, even when the late wake comes from another thread.
236        thread::spawn(move || late_waker.wake()).join().unwrap();
237        poll_pending_tasks();
238        assert_eq!(polls.get(), 2);
239        assert_eq!(*drops.lock().unwrap(), [owner]);
240    }

269    #[test]
270    fn duplicate_and_reentrant_wakes_are_polled_after_the_current_poll() {
271        let polls = Rc::new(Cell::new(0));
272        let observed_polls = polls.clone();
273        fdb_spawn(poll_fn(move |cx| {
274            let count = observed_polls.get() + 1;
275            observed_polls.set(count);
276            if count == 1 {
277                cx.waker().wake_by_ref();
278                cx.waker().wake_by_ref();
279                poll_pending_tasks();
280                assert_eq!(observed_polls.get(), 1);
281                Poll::Pending
282            } else {
283                Poll::Ready(())
284            }
285        }));
286        assert_eq!(polls.get(), 2);
287    }

289    #[test]
290    fn spawning_during_a_poll_defers_the_child_until_the_parent_returns() {
291        let events = Rc::new(RefCell::new(Vec::new()));
292        let observed_events = events.clone();
293        fdb_spawn(async move {

295            let child_events = observed_events.clone();
296            fdb_spawn(async move {
297                child_events.borrow_mut().push("child");

301        assert_eq!(*events.borrow(), ["parent enter", "parent exit", "child"]);
